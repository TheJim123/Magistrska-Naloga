\chapter{Hilbertovi prostori}

\section[Osnovne definicije in primeri]{Osnovne definicije in primeri}

	V tem poglavju bomo najprej osvežili znanje o nekaterih pomembnih definicijah in rezultatih iz študije Banachovih in Hilbertovih prostorov. V nadaljevanju bomo pozornost posvetili t.~i.~ adjungiranim operatorjem, nato pa še njihovim posebnim primerom - normalnim, sebi-adjungiranim in unitarnim operatorjem. Pri tem se bomo primarno sklicevali na vir \cite{bib:Rynne}.

\begin{definicija}
	\label{def:izometrija}
	Naj bosta $X$ ter $Y$ poljubna normirana prostora in naj bo $\map{T}{X}{Y}$ poljuben linearen operator. Pravimo, da je operator $T$ \pojem{izometrija}, če je $\norm{Tx}_Y = \norm{x}_X$ za vsak $x\in X$.
\end{definicija}

\begin{opomba}
	\label{opomb:izometrija}
	Kratek premislek nam pove, da je vsaka izometrija tudi omejen linearen operator.
\end{opomba}

\begin{izrek}[Heine-Borel-Lebesgue]
	\label{izr:HBL}
	Naj bo $\F^n$ poljuben metrični prostor, opremljen s standardno evklidsko metriko $d$. Za neprazno podmnožico $A\subseteq \F^n$ sta naslednji trditvi ekvivalentni: \begin{enumerate}[a)]
		\item $A$ je kompaktna,
		\item $A$ je zaprta in omejena.
	\end{enumerate}
\end{izrek}

\begin{lema}
	\label{lem:invnorm}
	Naj bosta $X$ ter $Y$ poljubna normirana prostora in naj bo $T\in B(X, Y)$ poljuben operator. Če je $T$ obrnljiv, potem za vsak $x\in X$ velja: $$\norm{Tx}\geq \norm{T^{-1}}^{-1}\norm{x}.$$
\end{lema}

\begin{lema}
	\label{lem:banzap}
	Naj bo $X$ poljuben Banachov prostor, $Y$ poljuben normiran prostor ter $T\in B(X, Y)$ poljuben omejen linearen operator. Če obstaja tak $r>0$, da je $\norm{Tx}\geq r\norm{x}$ za vsak $x\in X$, je $\operatorname{Im}T$ zaprti podprostor v $Y$.
\end{lema}

\begin{definicija}
	\label{def:odppresl}
	Naj bosta $(X, d_X)$ in $(Y, d_Y)$ poljubna metrična prostora. Pravimo, da je preslikava ~$\map{f}{X}{Y}$ \pojem{odprta}, če je za vsako odprto podmnožico $U$ v $X$ njena slika, $f(U)$, odprta podmnožica v $Y$.
\end{definicija}

\begin{izrek}[o odprti preslikavi]
	\label{izr:odppresl}
	Naj bosta $X$ in $Y$ poljubna Banachova prostora. Če je operator $T\in B(X, Y)$ surjektiven, je $T$ odprta preslikava.
\end{izrek}

\begin{posledica}
	\label{posl:odppresl}
	Naj bosta $X$ in $Y$ poljubna Banachova prostora. Če je $T\in B(X, Y)$ bijektiven operator, je~$T^{-1}\in B(Y, X)$.
\end{posledica}

\begin{izrek}[Hahn-Banach]
	\label{izr:HB} 
	Naj bo $(X, \norm{.})$ poljuben normiran prostor nad $\F$ in naj bo $Y$ poljuben podprostor prostora $X$. Za poljuben omejen linearen funkcional $\map{f}{Y}{\F}$, obstaja tak omejen linearen funkcional $\map{F}{X}{\F}$, da velja: \begin{enumerate}[a)]
		\item $F_{|_Y} = f$,
		\item $\norm{F} = \norm{f}$.
	\end{enumerate}
\end{izrek}

\begin{posledica}
	\label{posl:HB}
	Naj bo $X$ poljuben normiran prostor nad poljem $\F$. Tedaj za vsak element $x_0\in X\setminus\{0\}$ obstaja tak omejen linearen funkcional $\map{f}{X}{\F}$, da je: \begin{itemize}
		\item $f(x_0) = \norm{x_0}$,
		\item $\norm{f} = 1$.
	\end{itemize}
\end{posledica}

\begin{izrek}
	\label{izr:opHB} 
	Naj bo $(X, \norm{.})$ poljuben normiran prostor nad $\F$ in naj bo $Y$ poljubna gosta podmnožica prostora $X$. Naj bo $Z$ poljuben Banachov prostor nad $\F$. Tedaj za operator $T\in B(Y, Z)$ velja: \begin{enumerate}[a)]
		\item Naj bo $x\in X$ poljuben element in $\{x_n\}_{n\in\N},\{y_n\}_{n\in\N}$ poljubni zaporedji v $Y$, ki konvergirata k elementu $x$: $$\lim_{n\to \infty}x_n = x = \lim_{n\to \infty}y_n.$$ Tedaj sta tudi zaporedji $\{T(x_n)\}_{n\in\N}$ in $\{T(y_n)\}_{n\in\N}$ konvergentni in imata isto limito: $$\lim_{n\to \infty}Tx_n = \lim_{n\to \infty}Ty_n.$$
		\item Obstaja tak operator $U\in B(X, Z)$, da je $\norm{U} = \norm{T}$ in $Ux = Tx$ za vsak $x\in Y$.
	\end{enumerate}
\end{izrek}

\begin{definicija}
	\label{def:ScalProd}
	Naj bo $V$ vektorski prostor nad poljem $\F$. Preslikavi $\map{\Sp{.}{.}}{V\times V}{\R}$ pravimo \pojem{skalarni produkt} na $V$, če za poljubne elemente $x, x_1, x_2, y\in V$ ter poljubna skalarja $\alpha_1, \alpha_2\in \F$ velja: \begin{itemize}
		\item $\Sp{x}{x} \geq 0$,
		\item $\Sp{x}{x} = 0 \iff x = 0$,
		\item $\Sp{x}{y} = \overline{\Sp{y}{x}}$,
		\item $\Sp{\alpha_1\cdot x_1 + \alpha_2\cdot x_2}{y} = \alpha_1\Sp{x_1}{y} +\alpha_2\Sp{x_2}{y}$.
	\end{itemize}
	Če je $V$ vektorski prostor (nad $\F$) in $\Sp{.}{.}$ skalarni produkt na njem, pravimo, da je $(V, +, \cdot, \Sp{.}{.})$ \pojem{prostor s skalarnim produktom}.
\end{definicija}

\begin{primer}
	\label{prim:ScalProd}
	Spomnimo se nekaj znanih primerov vektorskih prostorov s skalarnim produktom: \begin{itemize}
		\item V prostoru $\R^n$ je s predpisom $\Sp{x}{y} = \displaystyle\sum_{i = 1}^{n}x_iy_i$ definiran skalarni produkt.
		\item V prostoru $\C^n$ je s predpisom $\Sp{z}{w} = \displaystyle\sum_{i = 1}^{n}z_i\overline{w_i}$ definiran skalarni produkt.
		\item V prostoru zaporedij $l^2$ je s predpisom $\Sp{\bar{x}}{\bar{y}} = \displaystyle\sum_{i = 1}^{\infty}x_i\overline{y_i}$ definiran skalarni produkt.
		\item V prostoru zveznih funkcij na zaprtem intervalu $[a, b]$, z oznako $\mathcal{C}([a, b])$, je s predpisom $\Sp{f}{g} = \int_{a}^{b}f(x)g(x)dx$ definiran skalarni produkt.
	\end{itemize}
\end{primer}

\begin{opomba}
	\label{opomb:ScalProd}
	Naj bo $(V, +, \cdot, \Sp{.}{.})$ prostor s skalarnim produktom nad poljem $\F$. Enostavno je preveriti, da je s predpisom $\norm{x} = \sqrt{\Sp{x}{x}}$ definirana norma na $V$. Za tako normo pravimo, da je \pojem{porojena s skalarnim produktom} $\Sp{.}{.}$.
\end{opomba}

\begin{izrek}[Neenakost Cauchy-Schwarz-Bunyakowsky]
	\label{izr:CSB}
	Naj bo $(X, \Sp{.}{.})$ poljuben vektorski prostor s skalarnim produktom nad poljem $\F$. Tedaj za vsak par elementov $x, y\in X$ velja: $$\abs{\Sp{x}{y}}^2 \leq \Sp{x}{x}\Sp{y}{y}.$$
\end{izrek}

\begin{opomba}
	\label{opomb:CSB}
	Če definiramo normo $\norm{.}$ tako, kot je v opombi \ref{opomb:ScalProd}, lahko neenakost v izreku \ref{izr:CSB} zapišemo tudi tako: $$\abs{\Sp{x}{y}}^2 \leq \norm{x}\cdot\norm{y}.$$
\end{opomba}

\begin{trditev}
	\label{trd:spzvez}
	Naj bo $X$ poljuben vektorski prostor s skalarnim produktom $\Sp{.}{.}$ nad poljem $\F$. Velja: \begin{enumerate}[a)]
		\item Preslikava $\Sp{.}{.}$ je zvezna preslikava iz $X\times X$ v $\F$.
		\item Če sta $\{x_n\}_{n\in\N}$ in $\{y_n\}_{n\in\N}$ konvergentni zaporedji v $X$ z limitama $x = \lim_{n\to \infty}x_n$ in $y = \lim_{n\to \infty}y_n$, potem je tudi zaporedje $\{\Sp{x_n}{y_n}\}_{n\in\N}$ konvergentno in velja: $$\Sp{x}{y} = \lim_{n\to \infty}\Sp{x_n}{y_n}.$$
	\end{enumerate}
\end{trditev}

\begin{definicija}
	\label{def:PolnMP}
	Naj bo $(M, d)$ poljuben metrični prostor. Pravimo, da je $M$ \pojem{poln} metrični prostor, če je vsako Cauchyjevo zaporedje v $M$ konvergentno v $M$.
\end{definicija}

\begin{primer}
	\label{prim:PolnMP}
	Navedimo nekaj klasičnih primerov polnih metričnih prostorov:
	\begin{itemize}
		\item prostori $\R^n$ s standardno Evklidsko normo;
		\item prostori $\C^n$ s standardno Evklidsko normo;
		\item prostor $l^2$ skupaj z normo, definirano s predpisom $\norm{\bar{x}}_2= \left(\displaystyle\sum_{i=1}^{\infty}\abs{x_i}^2\right)^\frac{1}{2}$;
		\item prostor s kvadratom integrabilnih funkcij na (realnem ali kompleksnem) merljivem prostoru $X$, ki je opremljen z Lebesguesovo mero $\mu$; $\mathcal{L}^2(X, \mu)$; skupaj z normo, definirano s predpisom $\norm{f}_2= \left(\displaystyle\int_{X}\abs{f}^2d\mu\right)^\frac{1}{2}$.
	\end{itemize}
\end{primer}

Sedaj definirajmo Hilbertove prostore.

\begin{definicija}
	\label{def:Hilbert}
	Naj bo $(V, \Sp{.}{.})$ poljuben prostor s skalarnim produktom nad poljem $\F$. Naj bo $\norm{.}$ norma na $V$ porojena s $\Sp{.}{.}$ in naj bo $\map{d}{V\times V}{\R}$ metrika na $V$ porojena z $\norm{.}$. Če je $(V, d)$ poln metrični prostor, pravimo, da je $(V, \Sp{.}{.})$ \pojem{Hilbertov prostor} nad $\F$. 
\end{definicija}

\begin{primer}
	\label{prim:Hilbert}
	Navedimo nekaj znanih primerov Hilbertovih prostorov. \begin{itemize}
		\item prostori $\R^n$ skupaj s standardnim skalarnim produktom;
		\item prostori $\C^n$ skupaj s (standardnim) skalarnim produktom iz primera \ref{prim:ScalProd};
		\item prostor $l^2$ skupaj s skalarnim produktom iz primera \ref{prim:ScalProd};
		\item prostor $\mathcal{L}^2(X, \mu)$ iz primera \ref{prim:PolnMP}, skupaj s skalarnim produktom, definiranim s predpisom $\Sp{f}{g}= \displaystyle\int_{[a, b]}(f\cdot \bar{g}) d\mu$.
	\end{itemize}
\end{primer}

\begin{opomba}
	\label{opomb:Hilbert}
	V primeru, ko je merljivi prostor $X$ iz primera \ref{prim:Hilbert} zaprti interval $[a, b]$, oznako za množico poenostavimo na $\mathcal{L}^2([a, b])$. Pri tem je mera na zaprtem intervalu kar standardna Lebesgueova mera. Predpis za skalarni produkt se v tem primeru spremeni v: $$\Sp{f}{g} = \int_{a}^{b}f(x)\overline{g(x)}dx.$$
\end{opomba}

\begin{primer}
	\label{prim:Tkop}
	Za vsako funkcijo $k\in \mathcal{C}([0, 1])$ definiramo operator $\map{T_k}{\mathcal{L}^2([0, 1])}{\mathcal{L}^2([0, 1])}$ na kompleksnem Hilbertovem prostoru $\mathcal{L}^2([0, 1])$ (s skalarnim produktom iz primera \ref{prim:Hilbert}) s predpisom: $$T_k(g) = k\cdot g.$$ Premislimo najprej, da je $T_k$ omejen linearen operator na $\mathcal{L}^2([0, 1])$. Najprej preverimo linearnost. Naj bosta $g_1, g_2\in \mathcal{L}^2([0, 1])$ poljubni funkciji ter $\alpha_1, \alpha_2\in\C$ poljubna skalarja. Tedaj je: \begin{align*}
		T_k(\alpha_1g_1 + \alpha_2g_2) &= k\cdot(\alpha_1g_1 + \alpha_2g_2) = k\cdot (\alpha_1g_1) + k\cdot (\alpha_2g_2) = \alpha_1(k\cdot g_1) + \alpha_2(k\cdot g_2) \\ &= \alpha_1T_k(g_1) + \alpha_2T_k(g_2).
	\end{align*}
	To pomeni, da je $T_k$ linearen operator. Sedaj preverimo še omejenost. Naj bo $g\in \mathcal{L}^2([0, 1])$ poljubna funkcija. Tedaj je: $$\norm{T_k(g)} = \left(\int_{0}^{1}\abs{k(t)g(t)}^2dt\right)^{\frac{1}{2}} = \left(\int_{0}^{1}\abs{k(t)}^2\abs{g(t)}^2dt\right)^{\frac{1}{2}}.$$ Ker je funkcija $k$ zvezna na intervalu $[0, 1]$, je tam tudi omejena, torej obstaja tak $M>0$, da je $\abs{k(t)}\leq M$ za vsak $t\in [0, 1]$. Posledično velja ocena: $$\norm{T_k(g)} = \left(\int_{0}^{1}\abs{k(t)}^2\abs{g(t)}^2dt\right)^{\frac{1}{2}} \leq \left(\int_{0}^{1}M^2\abs{g(t)}^2dt\right)^{\frac{1}{2}} = \left(M^2 \int_{0}^{1}\abs{g(t)}^2dt\right)^{\frac{1}{2}} = M \norm{g}.$$
	Ker je bila funkcija $g$ poljubna, sledi, da je operator $T_k$ omejen.
\end{primer}

\section[Ortogonalni komplement in Riezsov izrek]{Ortogonalni komplement in Riezsov izrek}

\begin{definicija}
	\label{def:ortogcompl}
	Naj bo $(X, \Sp{.}{.})$ poljuben vektorski prostor s skalarnim produktom nad poljem $\F$. Naj bo $A$ poljubna neprazna podmnožica prostora $X$. \pojem{Ortogonalni komplement} množice $A$ je množica $$A^\bot = \{x\in X~|~\Sp{x}{a}=0~\text{za vsak}~a\in A\}.$$
\end{definicija}

\begin{izrek}
	\label{izr:ortogdecomp}
	Naj bo $X$ poljuben Hilbertov prostor nad poljem $\F$. Tedaj za poljuben zaprti podprostor $Y$ velja, da je: $$X = Y \oplus Y^\bot.$$
\end{izrek}

\begin{posledica}
	\label{posl:zaportoggost}
	Naj bo $X$ poljuben Hilbertov prostor nad poljem $\F$. Tedaj za poljuben zaprti podprostor $Y$ velja, da je $Y^{\bot\bot} = Y$.
\end{posledica}

\begin{posledica}
	\label{posl:ortoggost}
	Naj bo $X$ poljuben Hilbertov prostor nad poljem $\F$. Tedaj za poljuben podprostor $Y$ velja, da je $Y^{\bot\bot} = \overline{Y}$.
\end{posledica}

\begin{izrek}[Riesz]
	\label{izr:Riesz}
	Naj bo $X$ poljuben Hilbertov prostor nad poljem $\F$. Potem za vsak linearen funkcional $\map{f}{X}{\F}$ obstaja enolično določen element $a\in X$, da je $f(x) = \Sp{x}{a}$ za vsak $x\in X$. Dodatno velja še, da je $\norm{f} = \norm{a}$.
\end{izrek}

\begin{lema}
	\label{lem:scalprodopenak}
	Naj bo $(X, \Sp{.}{.})$ poljuben kompleksen vektorski prostor s skalarnim produktom in naj bosta $S, T \in B(X)$ poljubna omejena linearna operatorja. Če je $\Sp{Tx}{x} = \Sp{Sx}{x}$ za vsak $x\in X$, je $T = S$.
\end{lema}

\begin{izrek}
	\label{izr:Hilbertopinv}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor. Naj bo $T\in B(X)$ poljuben omejen linearen operator. Če je $\norm{T}< 1$, je operator $I-T$ obrnljiv in velja: $$(I-T)^{-1} = \sum_{n = 1}^{\infty}T^n.$$
\end{izrek}

\begin{izrek}
	\label{izr:Hilbertinvset}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor ter naj bo $\mathcal{G}(X)$ množica vseh obrnljivih operatorjev v $B(X)$. Velja: \begin{enumerate}[a)]
		\item $(\mathcal{G}, \cdot)$ je grupa.
		\item Množica $\mathcal{G}$ je odprta v $B(X)$.
		\item Preslikava invertiranja, $\map{^{-1}}{\mathcal{G}}{\mathcal{G}}$, s predpisom $^{-1}(T) = T^{-1}$ je zvezna.
	\end{enumerate}
\end{izrek}

\begin{izrek}
	\label{izr:ortogbaza}
	Naj bo $X$ poljuben Hilbertov prostor nad poljem $\F$. Vsaka ortonormirana podmnožica prostora $X$ je vsebovana v neki maksimalni ortonormirani podmnožici prostora $X$.
\end{izrek}

\begin{izrek}
	\label{izr:fouriervrsta}
	Naj bo $X$ poljuben Hilbertov prostor nad poljem $\F$ in naj bo $M$ poljubna ortonormirana podmnožica v $X$. Tedaj za vsak $x\in X$ velja: \begin{enumerate}[a)]
		\item Množica $M_x = \{e\in M~|~ \Sp{x}{e}\neq 0\}$ je števna;
		\item $x = \displaystyle\sum_{e\in M_x} \Sp{x}{e}e$;
		\item $\norm{x}^2 = \displaystyle\sum_{e\in M_x}\abs{\Sp{x}{e}}^2$.
	\end{enumerate}
\end{izrek}

\chapter{Omejeni linearni operatorji na Hilbertovih prostorih}
V tem poglavju bomo obravnavali lastnosti omejenih linearnih operatorjev na Hilbertovih prostorih ter lastnosti njihovih spektrov, kadar sta domena in kodomena kompleksna Hilbertova prostora. Posebej nas bo zanimala operacija adjungiranja, zato bomo najprej povedali nekaj o njej.
\section[Adjungiran operator]{Adjungiran operator}

\begin{izrek}
	\label{izr:Adjexist}
	Naj bosta $(X, \Sp{.}{.}_X)$ in $(Y, \Sp{.}{.}_Y)$ poljubna kompleksna Hilbertova prostora in naj bo $T$ poljuben element $B(X, Y)$. Tedaj obstaja enolično določen operator $T^*\in B(Y, X)$, tak, da velja $$\Sp{Tx}{y}_Y = \Sp{x}{T^*y}_X$$ za vsak $x\in X$ in vsak $y \in Y$.
\end{izrek}

\begin{dokaz}
	Naj bo $y\in Y$ poljuben element ter naj bo $\map{f_y}{X}{\C}$ preslikava definirana s predpisom $f_y(x) = \Sp{Tx}{y}_Y$. Opazimo, da je $f_y$ linearen funkcional na $X$. Po izreku \ref{izr:CSB} potem za vsak element $x\in X$ velja: $$\abs{f_y(x)} = \abs{\Sp{Tx}{y}_Y} \leq \norm{Tx}_Y \cdot \norm{y}_Y \leq \norm{T}\cdot\norm{x}_X\cdot\norm{y}_Y.$$ To pomeni, da je $f_y$ omejen linearen funkcional na $X$. Posledično po Riezsovem izreku obstaja tak enolično določen $z\in X$, da je $f_y(x) = \Sp{x}{z}_X$ za vsak $x\in X$. Sedaj definiramo preslikavo $\map{T^*}{Y}{X}$, ki vsakemu $y\in Y$ priredi pripadajoči $z\in X$ (tisti, enolično določen, za katerega je $f_y(x) = \Sp{x}{z}_X$ za vsak $x\in X$). Posledično za vsak $x\in X$ in vsak $y\in Y$ velja: $$\Sp{Tx}{y}_Y=\Sp{x}{T^*y}_X .$$
	Sedaj preverimo, da je $T^*$ omejen linearen operator. Naj bodo $y_1, y_2 \in Y$ in $x\in X$ poljubni elementi ter $\alpha, \beta \in \C$ poljubna skalarja. Tedaj je \begin{align*}
		\Sp{x}{T^*(\alpha y_1 + \beta y_2)}_X &= \Sp{Tx}{\alpha y_1 + \beta y_2}_Y \\
		&= \bar{\alpha}\Sp{Tx}{y_1}_Y + \bar{\beta}\Sp{Tx}{y_2}_Y \\
		&= \bar{\alpha}\Sp{x}{T^*y_1}_X + \bar{\beta}\Sp{x}{T^*y_2}_X \\
		&= \Sp{x}{\alpha T^*y_1}_X + \Sp{x}{\beta T^*y_2}_X.
	\end{align*}
	To pomeni, da je $T^*(\alpha y_1 + \beta y_2)=\alpha T^*y_1 + \beta T^*y_2$ za vsak $x\in X$, vse $y_1, y_2 \in Y$ in vse $\alpha, \beta \in \C$, torej je $T^*$ res linearna preslikava. S pomočjo izreka \ref{izr:CSB} vidimo tudi, da velja $$\norm{T^*y}_X^2 = \Sp{T^*y}{T^*y}_X = \Sp{TT^*y}{y}_Y \leq \norm{TT^*y}_Y\cdot\norm{y}_Y \leq \norm{T}\cdot\norm{T^*y}_X\cdot\norm{y}_Y$$ za vsak $y\in Y$.
	Če je $\norm{T^*y}_X \neq 0$, dobljeno neenakost delimo z $\norm{T^*y}_X$ ter tako dobimo oceno $\norm{T^*y}_X \leq \norm{T}\cdot\norm{y}_Y$. V primeru, ko je $\norm{T^*y}_X = 0$, prejšnja ocena velja trivialno. Posledično je $$\norm{T^*y}_X \leq \norm{T}\cdot\norm{y}_Y$$ za vsak $y\in Y$.
	Preslikava $T^*$ je torej omejen linearen operator in velja, da je $\norm{T^*}\leq \norm{T}$. Za konec še pokažimo, da je $T^*$ enolično določen. Denimo, da imamo $U_1, U_2 \in B(Y, X)$, za katera velja, da je $\Sp{Tx}{y}_Y = \Sp{x}{U_1 y}_X = \Sp{x}{U_2 y}_X$ za vsak $x\in X$ in vsak $y\in Y$. Tedaj velja, da je $U_1y = U_2y$ za vsak $y\in Y$, torej je $U_1 = U_2$. To pomeni, da je $T^*$ res enolično določen.
\end{dokaz}
Izrek \ref{izr:Adjexist} nas motivira, da vpeljemo naslednjo definicijo.

\begin{definicija}
	\label{def:adjop}
	Naj bosta $X$ in $Y$ kompleksna Hilbertova prostora ter naj bo $T\in B(X, Y)$. Tedaj operatorju $T^*$ iz izreka \ref{izr:Adjexist} pravimo \pojem{adjungiran operator} operatorja $T$.
\end{definicija}
Začnimo z dvema enostavnima primeroma adjungiranih operatorjev. 
\begin{primer}
	\label{prim:trivadj}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor. 
	\begin{enumerate}[a)]
		\item Določimo adjunigiran operator ničelnega operatorja $\map{0}{X}{X}$. Vidimo, da za poljubna elementa $x, y\in X$ velja: $$\Sp{0x}{y} = \Sp{0}{y} = 0 = \Sp{x}{0} = \Sp{x}{0y}.$$ Posledično je $0^* = 0$.
		\item Določimo sedaj še adjungiran operator identičnega operatorja $\map{I}{X}{X}$. Opazimo, da velja: $$\Sp{Ix}{y} = \Sp{x}{y} = \Sp{x}{Iy}$$ za vse elemente $x, y\in X$. Posledično je $I^* = I$.
	\end{enumerate}
\end{primer}
Sedaj poglejmo malenkost zahtevnejša primera.
\begin{primer}
	\label{prim:adjop1} Definiramo operator $D$ na prostoru $l^2$, s predpisom $D(x_1, x_2, x_3, \ldots) = (0, x_1, x_2, \ldots)$. Ta operator imenujemo ">desni premik"<. Preverimo najprej, da je $D$ res omejen linearen operator na $l^2$. Vzemimo poljubni zaporedji $x, y\in l^2$ in poljubna skalarja $\alpha_1, \alpha_2\in\C$. Tedaj je: \begin{align*}
		D(\alpha_1 x + \alpha_2 y) &= D(\alpha_1 x_1 + \alpha_2 y_1 , \alpha_1 x_2 + \alpha_2 y_2, \ldots) = (0, \alpha_1 x_1 + \alpha_2 y_1 , \alpha_1 x_2 + \alpha_2 y_2, \ldots) \\ &= \alpha_1(0, x_1, x_2, \ldots) + \alpha_2(0, y_1, y_2, \ldots) = \alpha_1 Dx + \alpha_2 Dy.
	\end{align*}
	
	To pomeni, da je $D$ res linearen operator na $l^2$. Hitro opazimo tudi, da je operator $D$ izometrija. Vzemimo poljuben $x\in l^2$. Tedaj je: $$\norm{Dx} = \left(\sum_{i = 1}^{\infty}\abs{(Dx)_i}^2\right)^{\frac{1}{2}}=\left(0 + \abs{x_1}^2 + \abs{x_2}^2 +\ldots \right)^{\frac{1}{2}}=\norm{x}.$$
	To pomeni, da je $D$ omejen.
	
	Sedaj, ko vemo, da je $D\in B(l^2)$, določimo predpis njegovega adjungiranega operatorja $D^*$. Naj bosta $x, y\in l^2$ poljubna elementa ter označimo $z = D^*y$. Tedaj po definiciji adjungiranega operatorja velja, da je $\Sp{Dx}{y} = \Sp{x}{D^*y} = \Sp{x}{z}$ oziroma $$\Sp{(0, x_1, x_2, \ldots)}{(y_1, y_2, y_3, \ldots)}=\Sp{(x_1, x_2, x_3, \ldots)}{(z_1, z_2, z_3, \ldots)}.$$
	Ko to enakost razpišemo, dobimo: $$x_1\overline{y_2}+x_2\overline{y_3} + \ldots = x_1\overline{z_1} + x_2\overline{z_2}+x_3\overline{z_3} + \ldots.$$
	Opazimo, da če velja $z_i = y_{i+1}$ za vsak $i\in \N$, potem bo naša enačba veljavna za poljubno zaporedje $x$. Enoličnost adjungiranega operatorja nam posledično da predpis $$D^*y = D^*(y_1, y_2, y_3 \ldots) = (y_2, y_3, y_4, \ldots)$$ za vsak $y\in l^2$. Še več, prepoznamo, da je dobljeni operator ravno t.~i.~ ">levi premik"<, ki ga tipično označimo z $L$.
\end{primer}

\begin{primer}
	\label{prim:adjop2} 
	
	Iz primera \ref{prim:Tkop} vemo, da je operator $T_k$ omejen linearen operator na kompleksnem Hilbertovem prostoru $\mathcal{L}^2([0, 1])$. Sedaj bomo določili predpis njegovega adjungiranega operatorja $(T_k)^*$.
	
	Naj bosta $f, g \in \mathcal{L}^2([0, 1])$ poljubni funkciji ter naj bo $h = (T_k)^*g$. Tedaj po definiciji adjungiranega operatorja velja: $$\Sp{T_kf}{g} = \Sp{f}{(T_k)^*g} = \Sp{f}{h}.$$
	V skladu z opombo \ref{opomb:Hilbert} potem velja: $$\int_{0}^{1}k(t)f(t)\overline{g(t)}dt = \int_{0}^{1}f(t)\overline{h(t)}dt.$$
	Ta enačba bo držala, če je $\overline{h(t)} = k(t)\overline{g(t)}$ za vsak $t\in [0, 1]$ oziroma $h(t) = \overline{k(t)}g(t)$ za vsak $t\in [0, 1]$. Drugače povedano, enačba velja, če je $h = T_{\overline{k}}g$. Ker sta bili funkciji $f$ in $g$ poljubni, to pomeni, da sklep velja za vsak par funkcij iz $\mathcal{L}^2([0, 1])$. Ker je adjungirani operator enolično določen, sledi, da je $(T_k)^* = T_{\overline{k}}$.
\end{primer}

Sedaj bomo navedli in dokazali nekatere pomembne lastnosti adjungiranja.

\begin{izrek}
	\label{izr:adjlast}
	Naj bodo $(X, \Sp{.}{.}_X)$, $(Y, \Sp{.}{.}_Y)$ in $(Z, \Sp{.}{.}_Z)$ poljubni kompleksni Hilbertovi prostori, naj bodo $U, V \in B(X, Y)$ ter $T \in B(Y, Z)$ poljubni operatorji ter naj bosta $\alpha, \beta \in \C$ poljubna skalarja. Tedaj velja: \begin{enumerate}[a)]
		\item $(\alpha U + \beta V)^* = \bar{\alpha}U^* + \bar{\beta}V^*$,
		\item $(TU)^* = U^*T^*$,
		\item $(U^*)^* = U$,
		\item $\norm{U^*} = \norm{U}$,
		\item $\norm{U^*U} = \norm{U}^2$,
		\item Preslikava $\map{f}{B(X, Y)}{B(Y, X)}$, definirana s predpisom $f(U) = U^*$ je zvezna.
	\end{enumerate}
\end{izrek}

\begin{dokaz}
	\begin{enumerate}[a)]
		\item Vzemimo poljubna operatorja $U, V \in B(X, Y)$ in naj bosta $\alpha, \beta \in \C$ poljubna skalarja. Tedaj za vsak $x\in X$ in vsak $y\in Y$ velja: \begin{align*}
			\Sp{x}{(\alpha U + \beta V)^*y}_X &= \Sp{(\alpha U + \beta V)x}{y}_Y
			= \alpha \Sp{Ux}{y}_Y + \beta \Sp{Vx}{y}_Y \\
			&= \alpha \Sp{x}{U^*y}_X + \beta\Sp{x}{V^*y}_X
			= \Sp{x}{\bar{\alpha}U^*y}_X + \Sp{x}{\bar{\beta}V^*y}_X \\
			&= \Sp{x}{(\bar{\alpha} U^* + \bar{\beta} V^*)y}_X.
		\end{align*} 
		To pomeni, da je $(\alpha U + \beta V)^* = (\bar{\alpha} U^* + \bar{\beta} V^*)$.
		
		\item Naj bosta $U\in B(X, Y)$ in $T\in B(Y, Z)$ poljubna operatorja. Tedaj za vsak $x\in X$ in vsak $z\in Z$ velja: $$\Sp{x}{(TU)^*z}_X = \Sp{(TU)x}{z}_Z = \Sp{T(Ux)}{z}_Z = \Sp{Ux}{T^*z}_Y = \Sp{x}{U^*T^*z}_X.$$
		Posledično je res $(TU)^* = U^*T^*$.
		
		\item Vzemimo poljuben operator $U\in B(X, Y)$. Tedaj za vsak $x\in X$ in vsak $y\in Y$ velja: $$\Sp{y}{(U^*)^*x}_Y = \Sp{U^*y}{x}_X = \overline{\Sp{x}{U^*y}_X} = \overline{\Sp{Ux}{y}_Y} = \Sp{y}{Ux}_Y.$$
		Posledično je $(U^*)^* = U$.
		
		\item Naj bo $U\in B(X, Y)$ poljuben operator. V dokazu izreka \ref{izr:Adjexist} smo že pokazali, da je $\norm{U^*} \leq \norm{U}$. Ko upoštevamo prejšnjo točko, vidimo, da velja: $$\norm{U} = \norm{(U^*)^*} \leq \norm{U^*} \leq \norm{U}.$$
		Posledično je $\norm{U^*} = \norm{U}$.
		
		\item Vzemimo poljuben operator $U\in B(X, Y)$. Zaradi prejšnje točke tega dokaza vemo, da je $\norm{U^*} = \norm{U}$, hitro vidimo, da je $\norm{U^*U}\leq\norm{U^*}\cdot\norm{U} = \norm{U}^2$.
		Naj bo $x\in X$ poljuben element. Opazimo, da velja: $$\norm{Ux}_Y^2 = \Sp{Ux}{Ux}_Y = \Sp{U^*Ux}{x}_X.$$ Ko upoštevamo neenakost izrek \ref{izr:CSB}, dobimo naslednjo oceno: $$\Sp{U^*Ux}{x}_X \leq \norm{U^*Ux}_X\cdot\norm{x}_X \leq \norm{U^*U}\cdot\norm{x}_X^2.$$ Posledično je $\norm{Ux}_Y^2 \leq \norm{U^*U}\cdot\norm{x}_X^2$ za vsak $x\in X$. To pomeni, da je $\norm{U}^2 \leq \norm{U^* U}$. Od tod sledi iskana enakost.
		
		\item Naj bosta $U, V \in B(X, Y)$ poljubna operatorja in vzemimo poljubno število $\varepsilon > 0$. Izberemo $\delta = \varepsilon$ in denimo, da je $\norm{U - V} < \delta$. Tedaj je: $$\norm{f(U) - f(V)} = \norm{U^* - V^*} = \norm{(U - V)^*}.$$ Po točki $d)$ tega dokaza je $\norm{(U-V)^*} = \norm{U - V} < \delta = \varepsilon$, torej je preslikava $f$ zvezna.
	\end{enumerate}
\end{dokaz}

Pri obravnavi adjungiranih operatorjev bomo potrebovali naslednji pomožni rezultat.

\begin{lema}
	\label{lem:adjlast2}
	Naj bosta $(X, \Sp{.}{.}_X)$ in $(Y, \Sp{.}{.}_Y)$ poljubna kompleksna Hilbertova prostora. Tedaj za vsak operator $T\in B(X, Y)$ velja: \begin{enumerate}[a)]
		\item $\operatorname{Ker}T = (\operatorname{Im}T^*)^{\bot}$,
		\item $\operatorname{Ker}T^* = (\operatorname{Im}T)^{\bot}$,
		\item $\operatorname{Ker}^* = \{0\}$ natanko tedaj, ko je $\operatorname{Im}T$ gosta podmnožica v $Y$.
	\end{enumerate}
\end{lema}

\begin{dokaz}
	\begin{enumerate}[a)]
		\item Najprej pokažimo, da je $\operatorname{Ker}T \subseteq (\operatorname{Im}T^*)^\bot$. Vzemimo poljubna elementa $x\in \operatorname{Ker}T$ in $z\in \operatorname{Im}T^*$. Potem za $z$ obstaja tak $y\in Y$, da je $T^*y = z$. Posledično je $$\Sp{x}{z}_X = \Sp{x}{T^*y}_X = \Sp{Tx}{y}_Y = \Sp{0}{y}_Y = 0.$$
		Element $z\in\operatorname{Im}T^*$ je bil poljuben, torej je $x\in (\operatorname{Im}T^*)^\bot$. Ker je bil $x$ poljuben element $\operatorname{Ker}T$, je $\operatorname{Ker}T \subseteq (\operatorname{Im}T^*)^\bot$.
		Sedaj pokažimo še obratno inkluzijo. Naj bo $x\in (\operatorname{Im}T^*)^\bot$ poljuben element. Ker je $T^*Tx \in ImT^*$, velja: $$\norm{Tx}^2 = \Sp{Tx}{Tx}_Y = \Sp{x}{T^*Tx}_X= 0.$$
		To pomeni, da je $Tx = 0$ oz. $x\in \operatorname{Ker}T$. Ker je bil $x$ poljuben, je $(\operatorname{Im}T^*)^\bot \subseteq \operatorname{Ker}T$, od tod pa sledi iskana enakost.
		
		\item Da dokažemo ta rezultat, upoštevamo prejšnjo točko tega dokaza za $T^*$ ter točko $c)$ izreka \ref{izr:adjlast}. Posledično je $\operatorname{Ker}T^* = (\operatorname{Im}(T^*)^*)^\bot = (\operatorname{Im}T)^\bot$.
		
		\item Da dokažemo ta rezultat bomo dokazali, da veljata implikaciji v obe smeri.
		Najprej predpostavimo, da je $\operatorname{Ker}T^* = \{0\}$. Potem vidimo, upoštevajoč posledico \ref{posl:ortoggost} ter točko $b)$ tega dokaza, da velja: $$\overline{\operatorname{Im}T} = ((\operatorname{Im}T)^\bot)^\bot = (\operatorname{Ker}T^*)^\bot = \{0\}^\bot = Y.$$ To pomeni, da je $\operatorname{Im}T$ gosta podmnožica v $Y$.
		Dokažimo sedaj še obratno implikacijo. Denimo, da je $\operatorname{Im}T$ gosta podmnožica v $Y$. Potem po posledici \ref{posl:ortoggost} velja, da je $$((\operatorname{Im}T)^{\bot})^\bot = \overline{\operatorname{Im}T} = Y.$$ Po točki $b)$ tega dokaza velja tudi, da je $\operatorname{Ker}T^* = (\operatorname{Im}T)^\bot$. Ker je $\operatorname{Im}T\subseteq Y$ neprazna podmnožica, velja, da je $(\operatorname{Im}T)^\bot$ zaprti podprostor v $Y$. Po posledici \ref{posl:zaportoggost} potem sledi: $$\operatorname{Ker}T^* = (\operatorname{Im}T)^\bot =(((\operatorname{Im}T)^\bot)^\bot)^\bot = Y^\bot = \{0\}.$$
		
	\end{enumerate}
\end{dokaz}
S pomočjo lem \ref{lem:invnorm}, \ref{lem:banzap} in \ref{lem:adjlast2} dokažemo naslednjo posledico.
\begin{posledica}
	\label{posl:adjinv}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor ter naj bo $T\in B(X)$. Naslednji trditvi sta ekvivalentni: \begin{enumerate}[a)]
		\item Operator $T$ je obrnljiv.
		\item $\operatorname{Ker}T^* = \{0\}$ in obstaja tako število $r > 0$, da je $\norm{Tx} \geq r\norm{x}$ za vsak $x\in X$.
	\end{enumerate} 
\end{posledica}

\begin{dokaz}
	Dokazali bomo obe implikaciji.
	Predpostavimo najprej, da je operator $T\in B(X)$ obrnljiv. Tedaj je $T$ bijekcija in njegov inverz, $T^{-1}$, pripada $B(X)$. Posledično je $\operatorname{Im}T = X$. Zato je $\operatorname{Im}T$ gosta podmnožica v $X$ in tako nam lema \ref{lem:adjlast2} pove, da je $\operatorname{Ker}T^* = \{0\}$. Ker je $X$ Hilbertov prostor, zadošča tudi pogojem leme \ref{lem:invnorm}, ki nam pove, da velja $$\norm{Tx}\geq \norm{T^{-1}}^{-1}\norm{x}$$ za vsak $x\in X$. S tem smo dokazali prvo implikacijo, saj lahko za iskano število $r$ vzamemo kar $r = \norm{T^{-1}}^{-1}$.
	
	Sedaj pokažimo še obratno implikacijo. Denimo, da je $\operatorname{Ker}T^* = \{0\}$ in da obstaja tak $r>0$, da za vsak $x\in X$ velja ocena $\norm{Tx} \geq r\norm{x}$. Lema \ref{lem:adjlast2} nam potem pove, da je $\operatorname{Im}T$ gosta podmnožica v $X$, lema \ref{lem:banzap} pa, da je $\operatorname{Im}T$ zaprti podprostor v $X$. To pomeni, da je $X = \overline{\operatorname{Im}T} = \operatorname{Im}T$.
	Vzemimo sedaj poljuben element $x\in \operatorname{Ker}T$. Tedaj je $Tx = 0$, po predpostavki pa potem sledi, da je $0 = \norm{Tx} \geq r\norm{x}$. Slednje je lahko res le, kadar je $x = 0$. To pomeni, da je $\operatorname{Ker}T = \{0\}$, torej je $T$ bijektiven endomorfizem Banachovega prostora $X$. Po izreku o odprti preslikavi sledi, da je $T$ obrnljiv.
\end{dokaz}

\begin{primer}
	\label{prim:adjinv}
	Primer neobrnljivega operatorja najdemo v desnem premiku $D\in B(l^2)$. Kot smo premislili v primeru \ref{prim:adjop2}, je njegov adjungirani operator ravno levi premik $L\in B(l^2)$, za katerega pa velja, da je $L(1, 0, 0, \ldots) = \bar{0}$. Sledi, da $\operatorname{Ker}D^* = \operatorname{Ker}L \neq \{0\}$, torej $D$ ni obrnljiv.
\end{primer}

Za konec tega podpoglavja premislimo, v sledeči lemi, kako sta povezani operaciji invertiranja in adjungiranja.

\begin{lema}
	\label{lem:adjinv}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor. Če je operator $T\in B(X)$ obrnljiv, tedaj je obrnljiv tudi operator $T^*$. V tem primeru je $(T^*)^{-1} = (T^{-1})^*$.
\end{lema}

\begin{dokaz}
	Naj bo $T\in B(X)$ poljuben obrnljiv operator. Tedaj je $TT^{-1} = T^{-1}T = I$. Enačbo adjungiramo ter tako dobimo $(TT^{-1})^* = (T^{-1}T)^* = I^*$. Ko upoštevamo točko $b)$ izreka \ref{izr:adjlast}, se enačba poenostavi v: $$(T^{-1})^*T^* = T^*(T^{-1})^* = I.$$
	Posledično je $T^*$ obrnljiv ter $(T^*)^{-1} = (T^{-1})^*$.
\end{dokaz}

\section[Normalni operatorji]{Normalni operatorji}

Sedaj se bomo posvetili, glede na adjungiranje, posebnim primerom operatorjev. Prvi izmed teh, so t.~i.~ normalni operatorji.

\begin{definicija}
	\label{def:normalop}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor. Pravimo, da je operator $T \in B(X)$ \pojem{normalen}, če je $TT^* = T^*T$.
\end{definicija}
Poglejmo si nekaj primerov.
\begin{primer}
	\label{prim:normalop}
	Enostavno je videti, da je operator $T_k$, kot je bil definiran v primeru \ref{prim:Tkop}, normalen operator. Namreč za poljubno funkcijo $f\in \mathcal{L}^2([0, 1])$ velja:
	\begin{align*}
		(T_k(T_k)^*)f &= (T_kT_{\overline{k}})f =T_k(\overline{k}f) = k\overline{k}f~\text{in} \\
		((T_k)^*T_k)f &= (T_{\overline{k}}T_k)f = T_{\overline{k}}(kf)=\overline{k}kf = k\overline{k}f.
	\end{align*}
	Od tod sledi, da je $T_k(T_k)^* = (T_k)^*T_k$, torej je operator $T_k$ res normalen.
\end{primer}

\begin{primer}
	\label{prim:Tminlamnorm}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor, $T\in B(X)$ poljuben normalen operator in $\lambda\in\C$ poljuben skalar. Preverimo, da je operator $T-\lambda I$ tudi normalen operator. Namreč velja: \begin{align*}
		(T-\lambda I)(T - \lambda I)^* &= (T-\lambda I)(T^*-\bar{\lambda} I)
		= TT^* - \bar{\lambda}T -\lambda T^* + \lambda\bar{\lambda} I \\
		&= T^*T -\lambda T^* -\bar{\lambda}T + \bar{\lambda}\lambda I ~\text{in} \\
		(T-\lambda I)^*(T - \lambda I) &= (T^*-\bar{\lambda} I)(T-\lambda I)
		= T^*T -\lambda T^* -\bar{\lambda}T + \bar{\lambda}\lambda I.
	\end{align*}
	Sledi, da je $(T-\lambda I)(T - \lambda I)^* = (T-\lambda I)^*(T - \lambda I)$, torej je operator $(T-\lambda I)$ res normalen.
\end{primer}

\begin{primer}
	\label{prim:Dninorm}
	Ponovno se spomnimo operatorja $D$ na $l^2$. V primeru \ref{prim:adjop1} smo že določili njemu adjungiran operator, ki smo ga označili z $L$. Kratek račun bo pokazal, da operator $D$ ni normalen. Namreč za poljubno zaporedje $x = (x_1, x_2, \ldots)$ iz $l^2$ velja: \begin{align*}
		D^*Dx &= L(D(x_1, x_2, \ldots))= L(0, x_1, x_2, \ldots)=(x_1, x_2, \ldots)~\text{in} \\
		D(D^*)x &= D(L\Zap{x}) = D(x_2, x_3, \ldots) = (0, x_2, x_3, \ldots).
	\end{align*}
	Vidimo, da v splošnem ne velja $DD^*x = D^*Dx$. Za konkreten primer vzemimo zaporedje $e_1 = (1, 0, 0, \ldots)\in l^2$. Vidimo, da velja: \begin{align*}
		DD^*e_1 &= D0 = 0~\text{in} \\
		D^*De_1 &= D^*(0, 1, 0, \ldots) = e_1 \neq 0.
	\end{align*} To pomeni, da je $D^*D \neq DD^*$, torej operator $D$ ni normalen operator.
\end{primer}

\begin{lema}
	\label{lem:normalnorm}
	Naj bo $(X, \Sp{.}{.})$ poljuben kompleksen Hilbertov prostor. Naj bo $T\in B(X)$ poljuben normalen operator ter $r > 0$ poljubno realno število. Velja: \begin{enumerate}[a)]
		\item $\norm{Tx} = \norm{T^*x}$ za vsak $x\in X$.
		\item Če je $\norm{Tx} \geq r\norm{x}$ za vsak $x\in X$, je $\operatorname{Ker}T^* = \{0\}$.
	\end{enumerate}
\end{lema}

\begin{dokaz}
	\begin{enumerate}[a)]
		\item Naj bo $x\in X$ poljuben. Potem je \begin{align*}
			\norm{Tx}^2 - \norm{T^*x}^2 &= \Sp{Tx}{Tx} - \Sp{T^*x}{T^*x} = \Sp{T^*Tx}{x} - \Sp{TT^*x}{x}\\
			&= \Sp{T^*Tx - TT^*x}{x} = \Sp{(T^*T - TT^*)x}{x}.
		\end{align*}
		Ker je $T$ normalen, je $TT^* = T^*T$. Od tod sledi, da je $$\norm{Tx}^2 - \norm{T^*x}^2 = \Sp{(T^*T - TT^*)x}{x} = \Sp{0}{x} = 0.$$
		Posledično je $\norm{Tx} = \norm{T^*x}$. Ker je bil $x$ poljuben, enakost velja za vsak $x\in X$.
		\item Denimo, da obstaja tak $r>0$, da je $\norm{Tx} \geq r\norm{x}$ za vsak $x\in X$. Naj bo $x\in \operatorname{Ker}T^*$ poljuben. Po točki $a)$ tega dokaza je potem $0 = \norm{T^*x} = \norm{Tx} \geq r\norm{x}$. Sledi, da je $\norm{x} = 0$ oz. $x = 0$. Posledično je $\operatorname{Ker}T^* = \{0\}$.
	\end{enumerate}
\end{dokaz}

Na podlagi posledice \ref{posl:adjinv} in leme \ref{lem:normalnorm} sledi naslednja posledica.

\begin{posledica}
	\label{posl:normalinv}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor in naj bo $T\in B(X)$ poljuben normalen operator. Naslednji trditvi sta ekvivalentni: \begin{enumerate}[a)]
		\item Operator $T$ je obrnljiv.
		\item Obstaja tako realno število $r > 0$, da je $\norm{Tx} \geq r\norm{x}$ za vsak $x\in X$.
	\end{enumerate}
\end{posledica}

\begin{dokaz}
	Da dokažemo ekvivalenco, bomo dokazali implikaciji v obe smeri. Naj bo torej $X$ poljuben kompleksen Hilbertov prostor in $T\in B(X)$ poljuben normalen operator.
	Predpostavimo najprej, da je operator $T$ obrnljiv. Posledica \ref{posl:adjinv} nam pove, da obstaja tako realno število $r>0$, da je $\norm{Tx} \geq r\norm{x}$ za vsak $x\in X$. S tem je ta implikacija dokazana.
	
	Pokažimo sedaj še drugo implikacijo. Denimo torej, da obstaja tako realno število $r>0$, da je $\norm{Tx} \geq r\norm{x}$ za vsak element $x\in X$. Točka $b)$ leme \ref{lem:normalnorm} nam pove, da je potem $\operatorname{Ker}T^* = \{0\}$. Po posledici \ref{posl:adjinv} je potem operator $T$ obrnljiv.
\end{dokaz}

\section[Sebi-adjungirani operatorji]{Sebi-adjungirani operatorji}

Pri obravnavi preslikav nas pogosto zanimajo t.~i.~ ">fiksne točke"< - točke, ki jih preslikava preslika nazaj vase. V primeru adjungiranja, operatorjem, ki jih operacija ">fiksira"< damo posebno ime.

\begin{definicija}
	\label{def:sebiadjop}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor. Pravimo, da je operator $T\in B(X)$ \pojem{sebi-adjungiran}, če velja $T = T^*$.
\end{definicija}
Začnimo s trivialnima primeroma.
\begin{primer}
	\label{prim:sebiadjtriv}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor, $0$ ničelni operator na $X$ in $I$ identični operator na $X$. V primeru \ref{prim:trivadj} smo že premislili, da je $0^*=0$ in $I^* = I$. Vidimo, da sta oba operatorja sebi-adjungirana.
\end{primer}

\begin{opomba}
	\label{opomb:sebiadjsonormal}
	Naj bo $(X, \Sp{.}{.})$ poljuben kompleksen Hilbertov prostor in naj bo $T\in B(X)$ poljuben sebi-adjungiran operator na $X$. Enostavno je videti, da je potem $T$ tudi normalen operator: $$T^*T = TT = TT^*.$$
\end{opomba}

\begin{primer}
	\label{prim:sebiadjop}
	Vrnimo se k operatorju $T_k$ iz primera \ref{prim:adjop2} in premislimo, da je sebi-adjungiran za vsako realno funkcijo $k\in\mathcal{C}([0, 1])$. V tem primeru je namreč $\overline{k} = k$ in posledično je $(T_k)^* = T_{\overline{k}} = T_k$.
	
\end{primer}

V naslednji lemi so povzete nekatere osnovne lastnosti sebi-adjungiranih operatorjev.

\begin{lema}
	\label{lem:sebiadjlast}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor. Naj bo $\mathcal{S}(X)$ množica vseh sebi-adjungiranih operatorjev znotraj $B(X)$ in naj bo $T\in B(X)$ poljuben operator. Velja:
	\begin{enumerate}[a)]
		\item Za poljubni števili $\alpha, \beta \in \R$ in poljubna operatorja $T_1, T_2 \in \mathcal{S}(X)$ je $\alpha T_1 + \beta T_2 \in \mathcal{S}(X)$.
		\item Množica $\mathcal{S}(X)$ je zaprta podmnožica $B(X)$.
		\item Operatorja $TT^*$ in $T^*T$ sta sebi-adjungirana.
		\item $T = R + iS$ za neka sebi-adjungirana operatorja $R, S \in \mathcal{S}(X)$.
	\end{enumerate}
\end{lema}

\begin{dokaz}
	\begin{enumerate}[a)]
		\item Naj bosta $T_1, T_2 \in \mathcal{S}(X)$ poljubna operatorja. Ker sta operatorja sebi-adjungirana, po prvi točki izreka \ref{izr:adjlast} velja: $$(\alpha T_1 + \beta T_2)^* = \overline{\alpha} T_1^* + \overline{\beta} T_2^* = \alpha T_1 + \beta T_2.$$ Posledično je $\alpha T_1 + \beta T_2 \in \mathcal{S}(X)$.
		
		\item Naj bo $\{T_n\}_{n\in\N}$ konvergentno zaporedje s členi v $\mathcal{S}(X)$ in limito $T\in B(X)$. Zadnja točka izreka \ref{izr:adjlast} nam pove, da je adjungiranje zvezno, torej je tudi zaporedje $\{T_n^*\}_{n\in\N}$ konvergentno z limito $T^* \in B(X)$. Ker so $T_n \in \mathcal{S}(X)$ za vsak $n\in \N$, je $T_n^* = T_n$ za vsak $n\in \N$, torej je $T^* = T$. Posledično je tudi $T\in \mathcal{S}(X)$. Sledi, da je $\mathcal{S}(X)$ zaprta.
		
		\item Naj bo $T\in B(X)$ poljuben operator. Točka $c)$ izreka \ref{izr:adjlast} nam pove, da je $$(T^*T)^* = T^*T^{**} = T^*T.$$ Sledi, da je $T^*T\in \mathcal{S}(X)$. Na enak način premislimo, da enako velja za $TT^*$.
		
		\item Naj bo $T\in B(X)$ poljuben operator. Definiramo operatorja $R$ in $S$ na naslednji način: $$R = \frac{1}{2}(T + T^*)~\text{in}~ S=\frac{1}{2i}(T - T^*).$$
		Očitno velja, da je $T = R + iS$. Preverimo še, da sta $R$ in $S$ res sebi-adjungirana:
		\begin{align*}
			R^* &= (\frac{1}{2}(T + T^*))^* = \frac{1}{2}(T^* + T^{**}) = \frac{1}{2}(T^* + T) = R, \\
			S^* &= (\frac{1}{2i}(T - T^*))^* = \frac{-1}{2i}(T - T^*)^* = \frac{-1}{2i}(T^* - T^{**}) =\frac{1}{2i}(T - T^*) = S.
		\end{align*}
	\end{enumerate}
\end{dokaz}

\begin{opomba}
	\label{opomb:sebiadjlast}
	Ko upoštevamo, da je vsak zaprti podprostor Banachovega prostora tudi sam Banachov prostor, nam točki $a)$ in $b)$ leme \ref{lem:sebiadjlast} poveta, da je $\mathcal{S}(X)$ realen Banachov prostor.
\end{opomba}

\begin{trditev}
	\label{trd:sebiadjop}
	Naj bo $(X, \Sp{.}{.})$ poljuben kompleksen Hilbertov prostor. Operator $T\in B(X)$ je sebi-adjungiran natanko tedaj, ko je $\Sp{Tx}{x}\in\R$ za vsak $x\in X$.
\end{trditev}

\begin{dokaz}
	Dokazali bomo implikaciji v obe smeri.
		Predpostavimo najprej, da je operator $T$ sebi-adjungiran in vzemimo poljuben element $x\in X$. Tedaj je $\Sp{Tx}{x} = \Sp{x}{Tx} = \overline{\Sp{Tx}{x}}$, torej je $\Sp{Tx}{x}\in \R$. Ker je bil $x$ poljuben, sklep velja za vse elemente $X$.
		
		Denimo sedaj, da je $\Sp{Tx}{x}\in\R$ za vsak $x\in X$. Tedaj za poljubne skalarje $\alpha\in\C$ in poljubna elementa $x, y\in X$ velja, da je $\Sp{T(x + \alpha y)}{x + \alpha y}\in \R$ oziroma: $$\Sp{Tx}{x} + \alpha\Sp{Ty}{x} + \overline{\alpha}\Sp{Tx}{y} + \alpha\overline{\alpha}\Sp{Ty}{y}\in\R.$$
		Prvi in zadnji člen sta po predpostavki avtomatsko elementa $\R$. Sledi torej, da je tudi $\alpha\Sp{Ty}{x} + \overline{\alpha}\Sp{Tx}{y}\in\R$. Posledično velja: $$\alpha\Sp{Ty}{x} + \overline{\alpha}\Sp{Tx}{y} = \overline{\alpha}\overline{\Sp{Ty}{x}} + \overline{\overline{\alpha}}\overline{\Sp{Tx}{y}} = \overline{\alpha}\Sp{x}{Ty} + \alpha\Sp{y}{Tx}.$$
		Dodatno upoštevamo, da je $$\overline{\alpha}\Sp{x}{Ty} + \alpha\Sp{y}{Tx} = \overline{\alpha}\Sp{T^*x}{y} + \alpha\Sp{T^*y}{x}.$$ Sklepamo torej, da je $$\alpha\Sp{Ty}{x} + \overline{\alpha}\Sp{Tx}{y}=\overline{\alpha}\Sp{T^*x}{y} + \alpha\Sp{T^*y}{x}$$ za vsak $\alpha\in\C$. Če vstavimo $\alpha = 1$ in $\alpha = i$, dobimo naslednji sistem enačb: \begin{align*}
			\alpha = 1&:\Sp{Tx}{y} + \Sp{Ty}{x} = \Sp{T^*x}{y} + \Sp{T^*y}{x}, \\
			\alpha = i&:-i\Sp{Tx}{y} + i\Sp{Ty}{x} = -i\Sp{T^*x}{y} + i\Sp{T^*y}{x}.
		\end{align*}
		Prvo enačbo pomnožimo z $i$, jo odštejemo od druge, ter tako dobimo enakost $\Sp{Tx}{y} = \Sp{T^*x}{y}$. Ker sta bila $x$ in $y$ poljubna, ta sklep velja za vse pare $x, y \in X$. Zato po lemi \ref{lem:scalprodopenak} sledi, da je $T = T^*$.
\end{dokaz}

\section[Unitarni operatorji]{Unitarni operatorji}
			
Še en poseben primer operatorjev glede na adjungiranje je primer, v katerem je adjungirani operator hkrati tudi inverz. V tem poglavju bomo obravnavali le te.

\begin{definicija}
	\label{def:unitarop}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor. Pravimo, da je operator $T\in B(X)$ \pojem{unitaren}, če velja: $TT^* = T^*T = I$.
\end{definicija}
Enostaven premislek nam pokaže, da ničelni operator $0$ ni unitaren, identični operator $I$ pa je. Poleg tega trivialnega primera navedimo še enega.
\begin{primer}
	\label{prim:unitarop}
	Premislimo, da če funkcija $k\in\mathcal{C}([0, 1])$ zadošča pogoju $\abs{k(t)} = 1$ za vsak $t\in [0, 1]$, je operator $T_k$ (definiran v primeru \ref{prim:adjop2}) unitaren operator.
	V primeru \ref{prim:normalop} smo že premislili, da je za poljuben $k\in\mathcal{C}([0, 1])$ operator $T_k$ normalen ter da je $$T_k(T_k)^*f = k\overline{k}f =\abs{k}^2f= (T_k)^*T_kf$$ za vsako funkcijo $f\in\mathcal{L}^2([0, 1])$. Vzemimo torej poljubno funkcijo $f\in\mathcal{L}^2([0, 1])$ in naj bo funkcija $k\in\mathcal{C}([0, 1])$ taka, da je $\abs{k(t)}=1$ za vsak $t\in[0, 1]$. Tedaj je $\abs{k(t)}^2f(t) = f(t)$ za vsak $t\in[0, 1]$, torej je $$T_k(T_k)^*f = f = (T_k)^*T_kf.$$ Ker je bil $f$ poljuben, sledi, da je $T_k(T_k)^* = I = (T_k)^*T_k$. To pomeni, da je operator $T_k$ je unitaren.
\end{primer}

Naslednji izrek nam pove, da so unitarni operatorji natanko surjektivne izometrije.

\begin{izrek}
	\label{izr:uniizo}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor. Za poljubna operatorja $T, U\in B(X)$ velja: \begin{enumerate}[a)]
		\item $T^*T = I$ natanko tedaj, ko je $T$ izometrija.
		\item $U$ je unitaren operator natanko tedaj, ko je $U$ surjektivna izometrija na $X$.
	\end{enumerate}
\end{izrek}

\begin{dokaz}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor in vzemimo poljubna operatorja $T, U\in B(X)$.
	\begin{enumerate}[a)]
		\item Predpostavimo najprej, da je $T^*T = I$ in vzemimo poljuben element $x\in X$. Tedaj je: $$\norm{Tx}^2=\Sp{Tx}{Tx} = \Sp{T^*Tx}{x}= \Sp{Ix}{x} = \Sp{x}{x} = \norm{x}^2.$$
		Ker je bil $x$ poljuben, zgornja enakost velja za vse elemente $X$, torej je $T$ izometrija.
			
		Denimo sedaj, da je $T$ izometrija in vzemimo poljuben element $x\in X$. Tedaj velja: $$\Sp{T^*Tx}{x} = \Sp{Tx}{Tx} = \norm{Tx}^2 = \norm{x}^2 = \Sp{Ix}{x}.$$
		Lema \ref{lem:scalprodopenak} nam pove, da je potem $T^*T = I$.
			
		\item Pri dokazu te ekvivalence, si bomo pomagali s prejšnjo točko dokaza.
		Predpostavimo najprej, da je operator $U$ unitaren. To pomeni, da je $UU^*=U^*U = I$ in točka $a)$ tega dokaza nam pove, da je $U$ izometrija. Vzemimo sedaj poljuben element $x\in X$. Tedaj je $x = U(U^*x)$, torej je $x\in \operatorname{Im}U$. To pomeni, da je $\operatorname{Im}U = X$. Posledično je operator $U$ surjektiven.
			
		Denimo sedaj, da je operator $U$ surjektivna izometrija na $X$. Točka $a)$ tega dokaza nam pove, da je potem $U^*U = I$. Vzemimo sedaj poljuben element $y\in X$. Ker je $U$ surjektiven operator na $X$, obstaja tak $x\in X$, da je $Ux = y$. Posledično velja: $$UU^*y = UU^*(Ux) = U(U^*Ux) = U(Ix) = Ux = y.$$
		Ker je bil $y$ poljuben, prejšnji sklep velja za vsak element $X$. To pomeni, da je $UU^* = I$. Sledi, da je $U$ unitaren operator.
	\end{enumerate}
\end{dokaz}

Da dopolnimo znanje o unitarnih operatorjih, premislimo še naslednjo lemo.

\begin{lema}
	\label{lem:unitarset}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor ter naj bo $\mathcal{U}(X)$ množica vseh unitarnih operatorjev na $X$. Velja:
	\begin{enumerate}[a)]
		\item Če je $U\in \mathcal{U}(X)$, je tudi $U^*\in \mathcal{U}(X)$ ter $\norm{U} = \norm{U^*} = 1$.
		\item Če sta $U_1, U_2 \in \mathcal{U}(X)$, sta tudi $U_1U_2$ in $U_1^{-1}$ unitarna operatorja.
		\item $\mathcal{U}(X)$ je zaprta podmnožica $B(X)$.
	\end{enumerate}
\end{lema}

\begin{dokaz}
	\begin{enumerate}[a)]
		\item Naj bo $U$ poljuben unitaren operator na $X$. Tedaj je $$U^*(U^*)^* = U^*U = I$$ in $$(U^*)^*U^* = UU^* = I.$$ To pomeni, da je tudi operator $U^*\in B(X)$ unitaren. Po izreku \ref{izr:adjlast} že vemo, da je $\norm{U} = \norm{U^*}$. Hkrati nam izrek \ref{izr:uniizo} pove, da je $U$ izometrija, torej je $\norm{U} = 1$.
		\item Naj bosta $U_1$ in $U_2$ poljubna unitarna operatorja na $X$. S pomočjo izreka \ref{izr:adjlast} vidimo, da je: \begin{align*}
			U_1U_2(U_1U_2)^* &= U_1U_2U_2^*U_1^* = U_1IU_1^* = I, \\
			(U_1U_2)^*U_1U_2 &= U_2^*U_1^*U_1U_2 = U_2^*IU_2 = I.
		\end{align*} To pomeni, da je tudi $U_1U_2 \in \mathcal{U}(X)$.
		Ker je $U_1$ unitaren, je omejen in po izreku \ref{izr:uniizo} bijektiven. Zato je po posledici \ref{posl:odppresl} tudi obrnljiv (znotraj $B(X)$). Dodatno upoštevamo lemo \ref{lem:adjinv} in tako dobimo: \begin{align*}
			(U_1^{-1})^*U_1^{-1} &= (U_1^*)^{-1}U_1^{-1} = (U_1U_1^*)^{-1} = I^{-1} = I, \\
			U_1^{-1}(U_1^{-1})^* &= U_1^{-1}(U_1^*)^{-1} = (U_1^*U_1)^{-1} = I^{-1} = I.
		\end{align*}
		Sledi, da je $U_1^{-1}$ unitaren.
		\item Naj bo $\{U_n\}_{n\in\N}$ zaporedje s členi v $\mathcal{U}(X)$ in limito $U \in B(X)$. Potem je po točki $a)$ tega dokaza tudi zaporedje $\{U_n^*\}_{n\in\N}$ zaporedje s členi v $\mathcal{U}(X)$. Ker je adjungiranje zvezno, je $U^*\in B(X)$ limita zaporedja $\{U_n^*\}_{n\in\N}$. Ker je $U$ limita zaporedja $\{U_n\}_{n\in\N}$, je $Ux = \lim_{n\to\infty}U_nx$ za vsak element $x\in X$. Vzemimo sedaj poljuben element $x\in X$. Ker je norma zvezna preslikava, velja: $$\norm{Ux} = \norm{\lim_{n\to\infty}U_nx} = \lim_{n\to\infty}\norm{U_nx}.$$ Po točki $b)$ izreka \ref{izr:uniizo} je operator $U_n$ izometrija za vsak $n\in\N$, torej je $\norm{U_nx} = \norm{x}$ za vsak $n\in\N$. To pomeni, da je $$\norm{Ux} = \lim_{n\to\infty}\norm{U_nx} = \lim_{n\to\infty}\norm{x} = \norm{x}.$$ Sledi, da je $U$ izometrija. Zato po točki $a)$ izreka \ref{izr:uniizo} sledi, da je $U^*U = I$. Na enak način kot za $U$ vidimo, da je tudi $U^*$ izometrija in posledično po izreku \ref{izr:uniizo} velja, da je $(U^*)^*U^* = UU^* = I$. Sledi, da je $U$ unitaren.
	\end{enumerate}
\end{dokaz}

\section[Ortogonalni projektorji]{Ortogonalni projektorji}
Zadnji poseben tip operatorjev glede na adjungiranje so t.~i.~ ">ortogonalni projektorji"<. Podajmo najprej definicijo.
\begin{definicija}
	\label{def:ortproj}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor. Pravimo, da je operator $P\in B(X)$ \pojem{ortogonalen projektor}, če je: $$P^2 = P = P^*.$$
\end{definicija}
Izkaže se, da imajo ortogonalni projektorji zanimivo zvezo z zaprtimi podprostori poljubnega danega kompleksnega Hilbertovega prostora ter z njegovim ortogonalnim razcepom. To zvezo opišemo v naslednjem izreku.
\begin{izrek}
	\label{izr:ortogprojdekomp}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor. Velja: \begin{enumerate}
		\item Za poljuben zaprt podprostor $Y$ v $X$ obstaja tak ortogonalen projektor $P_Y\in B(X)$, da je $\operatorname{Im}P_Y = Y$, $\operatorname{Ker}P_Y = Y^\bot$ in $\norm{P_Y} \leq 1$.
		\item Za poljuben ortogonalen projektor $Q\in B(X)$ je $\operatorname{Im}Q$ zaprt podprostor v $X$ in velja: $$Q = P_{\operatorname{Im}Q}.$$
	\end{enumerate}
\end{izrek}

\begin{dokaz}
	\begin{enumerate}
		\item Naj bo $Y$ poljuben zaprti podprostor v $X$. Spomnimo se, da lahko tedaj $X$ zapišemo kot direktno vsoto podprostorov: $$X = Y \oplus Y^\bot.$$ Vzemimo torej poljuben element $x\in X$. Vemo, da tedaj obstajata enolično določena elementa $y\in Y$ in $z\in Y^\bot$, da je $x = y + z$. Sedaj definiramo preslikavo $\map{P_Y}{X}{X}$ s predpisom $P_Y(x) = y$. Pokazali bomo, da je ta preslikava zadošča vsem pogojem trditve.
		
		Premislimo najprej, da je preslikava $P_Y$ linearna. Vzemimo poljubna elementa $x_1, x_2\in X$ in poljubna skalarja $\lambda_1, \lambda_2\in \C$. Naj bodo $y_1, y_2\in Y$ in $z_1, z_2\in Y^\bot$ tisti elementi, za katere je $x_1 = y_1 + z_1$ in $x_2 = y_2 + z_2$. Izberemo take elemente $y_1, y_2, z_1$ in $z_2$. Tedaj je $$\lambda_1x_1 + \lambda_2x_2 = \lambda_1(y_1 + z_1) + \lambda_2(y_2 + z_2) = (\lambda_1y_1 + \lambda_2y_2) + (\lambda_1z_1 + \lambda_2z_2).$$ Ker sta $Y$ in $Y^\bot$ oba vektorska prostora sledi: $$\lambda_1y_1 + \lambda_2y_2\in Y~\text{in}~\lambda_1z_1 + \lambda_2z_2\in Y^\bot.$$ Posledično je $$P_Y(\lambda_1x_1 + \lambda_2x_2)= \lambda_1y_1 + \lambda_2y_2 = \lambda_1P_Y(x_1) + \lambda_2P_Y(x_2).$$ To pomeni, da je $P_Y$ res linearen operator na prostoru $X$.
		
		Kratek račun pokaže, da je operator $P_Y$ tudi omejen. Vzemimo spet poljuben element $x\in X$ in naj bosta elementa $y\in Y$ ter $z\in Y^\bot$ tista, za katera je $x = y + z$. Tedaj je $$\norm{P_Y(x)} = \norm{y} \leq \norm{x}.$$ To pomeni, da je opertor $P_Y$ omejen in da je $\norm{P_Y}\leq 1$.
		
		Premislimo sedaj, da je operator $P_Y$ sebi-adjungiran. Naj bosta $x_1, x_2\in X$ poljubna elementa. Naj bodo $y_1, y_2\in Y$ ter $z_1, z_2\in Y^\bot$ tisti elementi, za katere je je $x_1 = y_1 + z_1$ in $x_2 = y_2 + z_2$. Potem je $$\Sp{P_Y(x_1)}{x_2} = \Sp{y_1}{y_2 + z_2} = \Sp{y_1}{y_2}$$ in $$\Sp{x_1}{P_Y(x_2)} = \Sp{y_1+ z_1}{y_2} = \Sp{y_1}{y_2}.$$ Sledi, da je $\Sp{P_Y(x_1)}{x_2} = \Sp{x_1}{P_Y(x_2)}$ za vsaka elementa $x_1, x_2 \in X$. To pomeni, da je operator $P_Y$ sebi-adjungiran.
		
		Sedaj preverimo, da je $\operatorname{Im}P_Y = Y$. Naj bo $y\in Y$ poljuben element. Tedaj je ortogonalni razcep tega elementa enak $y = y + 0$, torej je $P_Y(y) = y$. To pomeni, da je $Y\subseteq \operatorname{Im}P_Y$. Hkrati nam predpis operatorja $P_Y$ pove, da je $\operatorname{Im}P_Y \subseteq Y$. Torej je $\operatorname{Im}P_Y = Y$. Točka $a)$ leme \ref{lem:adjlast2} nam tudi pove, da je: $$\operatorname{Ker}P_Y = (\operatorname{Im}(P_Y)^*)^\bot = (\operatorname{Im}P_Y)^\bot = Y^\bot.$$
		
		Za konec preverimo, da je operator $P_Y$ ortogonalen projektor. Naj bo $x\in X$ poljuben element in naj bosta elementa $y\in Y$ in $z\in Y^\bot$ tista, za katera je $x = y + z$. Tedaj je: $$P_Y^2(x)=P_Y(P_Y(x))=P_Y(y)=y=P_Y(x).$$ To pomeni, da je $P_Y^2 = P_Y$ in ker je operator $P_Y$ sebi-adjungiran, sledi, da je tudi ortogonalen projektor.
		
		\item Naj bo $Q\in B(X)$ poljuben ortogonalen projektor in označimo $H = \operatorname{Im}Q$. Ker je $Q$ linearen operator na $X$, je $H$ linearen podprostor prostora $X$. Naj bo sedaj $\{y_n\}_{n\in\N}$ poljubno konvergentno zaporedje v $H$, ki konvergira k točki $y\in X$. Ker je za vsak $n\in\N$ element $y_n\in \operatorname{Im}Q$, sledi, da obstajajo taki elementi $x_n\in X$, da je $Q(x_n) = y_n$ za vsak $n\in\N$. Ko dodatno upoštevamo, da je operator $Q$ omejen in da je $Q^2 = Q$, vidimo naslednje:
		$$y = \lim_{n\to \infty}Q(x_n) = \lim_{n\to \infty}Q^2(x_n) = Q(\lim_{n\to \infty}Q(x_n)) = Q(\lim_{n\to \infty}y_n) = Q(y).$$ To pomeni, da je $y\in H$ oz. podprostor $H$ je zaprt v $X$.
		
		Pokažimo sedaj, da je $Q = P_{\operatorname{Im}Q}$. Vzemimo najprej poljuben element $y\in H$. Tedaj obstaja tak element $x\in X$, da je $y = Q(x)$. To pomeni, da je: $$Q(y) = Q(Q(x)) = Q^2(x) = Q(x).$$
		Vzemimo sedaj poljuben element $z\in H^\bot$. Spomnimo se, da je operator $Q$ sebi-adjungiran in velja, da je $Q^2(z)\in H$. Zato sledi, da je: $$\norm{Q(z)}^2 = \Sp{Q(z)}{Q(z)} = \Sp{z}{Q^2(z)} = 0.$$
		
		Vzemimo sedaj poljuben element $x\in X$ ter naj bosta $y\in H$ in $z\in H^\bot$ tista elementa, za katera je $x = y + z$. Potem je $x = Q(y) + z$. Na enak način kot v točki $a)$ tega dokaza definiramo ortogonalni projektor $P_H$. Ker je $Q(z)=0$, je $P_H(x) = Q(y) = Q(x)$. Ker je bil element $x\in X$ poljuben, to pomeni, da je $P_H = Q$ oziroma $P_{\operatorname{Im}Q} = Q$.
	\end{enumerate}
\end{dokaz}
Naravno se pojavi vprašanje, ali obstaja kakšna zveza med ortogonalnima projektorjema $P_Y$ in $P_{Y^\bot}$. Odgovor na to vprašanje je pritrdilen, kar bomo pokazali v naslednji lemi.
\begin{lema}
	\label{lem:ortogproj2}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor in $Y$ poljuben zaprt podprostor v $X$. Če je $P\in B(X)$ ortogonalen projektor prostora $X$ na $Y$, potem je operator $I - P$ ortogonalen projektor prostora $X$ na $Y^\bot$.
\end{lema}

\begin{dokaz}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor in $Y$ poljuben zaprt podprostor v $X$. Naj bo $P\in B(X)$ poljuben ortogonalen projektor prostora $X$ na $Y$. Ker sta identični operator $I$ in operator $P$ oba sebi-adjungirana, je tudi operator $I-P$. Dodatno, ker je $P^2 = P$, velja: $$(I-P)^2 = (I-P)(I-P) = I^2 - 2P + P^2 = I - 2P + P = I-P.$$ To pomeni, da je operator $I-P$ ortogonalni projektor.
	Vzemimo sedaj poljuben element $x\in X$. Naj bosta elementa $y\in Y$ in $z\in Y^\bot$ tista, za katera je $x = y + z$. Tedaj je $P(x) = y$ in $(I-P)(x) = x - y = z$. To pomeni, da je operator $I-P$ ortogonalni projektor prostora $X$ na $Y^\bot$.
\end{dokaz}
S pomočjo izreka \ref{izr:ortogprojdekomp} premislimo naslednjo posledico.
\begin{posledica}
	\label{posl:ortprojsum}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor in $Y$ poljuben zaprt podprostor v $X$. Naj bo $B$ poljubna ortonormirana baza podprostora $Y$. Tedaj za ortogonalen projektor $P\in B(X)$ prostora $X$ na $Y$ velja: $$Px = \sum_{e\in B}\Sp{x}{e}\cdot e$$ za vsak $x\in X$.
\end{posledica}

\begin{dokaz}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor in $Y$ poljuben zaprt podprostor v $X$. Vzemimo poljubno ortonormirano bazo $B$ podprostora $Y$. Bazo $B$ lahko razširimo do ortonormirane baze $B' = B \cup B''$ celotnega prostora $X$. Ker je $B$ baza za $Y$ in je $X = Y \oplus Y^\bot$ sledi, da bo $B''$ tvorila bazo za podprostor $Y^\bot$. Naj bo sedaj $P\in B(X)$ poljuben ortogonalen projektor prostora $X$ na $Y$ in vzemimo poljuben element $x\in X$. Naj bosta $y\in Y$ in $z\in Y^\bot$ tista elementa, za katera je $x = y + z$. Velja, da je $\Sp{y}{e''} = 0$ za vsak $e'' \in B''$ ter da je $\Sp{z}{e}=0$ za vsak $e\in B$. Sedaj elementa $y$ in $z$ razvijemo po bazi $B'$: $$y = \sum_{e\in B'}\Sp{y}{e}e=\sum_{e\in B}\Sp{y}{e}e = \sum_{e\in B}\Sp{y+z}{e}e = \sum_{e\in B}\Sp{x}{e}e$$ in $$z = \sum_{e\in B'}\Sp{z}{e}e=\sum_{e\in B''}\Sp{z}{e}e = \sum_{e\in B''}\Sp{y+z}{e}e = \sum_{e\in B''}\Sp{x}{e}e.$$
	Od tod sledi, da je: $$P(x) = P(y + z) = y = \sum_{e\in B}\Sp{x}{e}e.$$
\end{dokaz}

Ortogonalni projektorji na Hilbertovih prostorih nas motivirajo tudi, da začnemo obravnavati vprašanje, ali za vsak izbran pozitiven operator $T$ na poljubnem Hilbertovem prostoru obstaja tak operator $U$ na istem prostoru, da je $U^2 = T$. 

\begin{definicija}
	\label{def:opkoren}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor in $T\in B(X)$ poljuben operator. \pojem{Kvadratni koren} operatorja $T$ je tak operator $U\in B(X)$, da je $U^2 = T$.
\end{definicija}

Da bi povedali kaj več na to temo, moramo najprej uvesti pojem spektra operatorja. To bomo storili v naslednjem poglavju.

\chapter{Spekter}
Problem lastnih vrednosti matrik nam je znan že iz linearne algebre, ni pa omejen le na matrike - na enak način jih lahko definiramo tudi za operatorje na poljubnih prostorih. V tem poglavju bomo obravnavali t.~i.~ spekter operatorja na kompleksnem Hilbertovem prostoru.

\begin{definicija}
	\label{def:specter}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor ter naj bo $T\in B(X)$ poljuben operator. \pojem{Spekter} operatorja $T$ je množica, definirana na naslednji način: $$\sigma(T) = \{\lambda\in\C~|~\text{operator}~T-\lambda I~\text{ni obrnljiv}\}.$$
\end{definicija}

\begin{opomba}
	\label{opomb:specter}
	Direktno iz definicije spektra sledi naslednja opazka: Operator $T\in B(X)$ ni obrnljiv natanko tedaj, ko je $0\in\sigma(T)$.
\end{opomba}

\begin{primer}
	\label{prim:trivspec}
	Določimo spekter za operatorja $0$ in $I$. Naj bo $\lambda\in\C$ poljuben. Določimo najprej spekter za $0$. Denimo, da je $\lambda\neq 0$. Potem je operator $0-\lambda I = -\lambda I$ obrnljiv in njegov inverz je $-\lambda^{-1}I$. Če je pa $\lambda = 0$, operator $-\lambda I = 0$ ni obrnljiv. Posledično je $\sigma(0)=\{0\}$. 
	Določimo sedaj še spekter za $I$. Operator $I - \lambda I = (1-\lambda)I$ je obrnljiv v primeru, ko je $1-\lambda \neq 0$ oz. ko je $\lambda \neq 1$. Res, tedaj je njegov inverz ravno operator $(1-\lambda)^{-1}I$. V primeru, ko je $\lambda=1$, operator $(1-\lambda)I = 0$ ni obrnljiv. To pomeni, da je $\sigma(I) = \{1\}$.
\end{primer}

V naslednji lemi preverimo, da spekter vsebuje vse lastne vrednosti operatorja.

\begin{lema}
	\label{lem:eigeninspecter}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor. Če je $\lambda$ lastna vrednost operatorja $T\in B(X)$, potem je $\lambda\in\sigma(T)$.
\end{lema}

\begin{dokaz}
	Naj bo $T\in B(X)$ poljuben ter naj bo $\lambda$ njegova lastna vrednost. Tedaj obstaja nek element $x\in X\setminus\{0\}$, za katerega velja: $$Tx = \lambda x.$$ To pomeni, da je $x\in \operatorname{Ker}(T-\lambda I)$, torej operator $T - \lambda I$ ni injektiven in posledično ni obrnljiv. Sledi, da je $\lambda\in\sigma(T)$.
\end{dokaz}

\begin{opomba}
	\label{opomb:eigeninspecter}
	V primeru končno razsežnih vektorskih prostorih vemo, da je linearna preslikava injektivna natanko teda, ko je surjektivna. Posledično, je spekter linearne preslikave na končno razsežnem kompleksnem Hilbertovem prostoru natanko množica njenih lastnih vrednosti. V splošnem to ni res. Obstajajo namreč omejeni linearni operatorji na (neskončno razsežnih) kompleksnih Hilbertovih prostorih, ki imajo neprazen spekter, a hkrati ne premorejo nobene lastne vrednosti. Primer takega operatorja je ravno desni premik $D\in B(l^2)$ iz primera \ref{prim:adjop1}. Da je spekter vedno neprazna množica, bomo premislili v izreku \ref{izr:specter}.
\end{opomba}

Preden določimo nekaj osnovnih lastnosti spektra operatorja, se spomnimo Liouvilleovega izreka iz kompleksne analize.

\begin{izrek}[Liouvilleov]
	\label{izr:Liouville}
	Naj bo $\map{f}{\C}{\C}$ poljubna funkcija, ki je (kompleksno) odvedljiva na celotni kompleksni ravnini $\C$. Če je $f$ omejena, je tudi konstantna.
\end{izrek}

\begin{izrek}
	\label{izr:specter}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor. Za vsak operator $T\in B(X)$ velja: \begin{enumerate}[a)]
		\item Če je $\lambda\in\sigma(T)$, je $\abs{\lambda}\leq\norm{T}$.
		\item $\sigma(T)$ je kompaktna množica v $\C$.
		\item $\sigma(T)\neq \emptyset$.
	\end{enumerate}
\end{izrek}

\begin{dokaz}
	Tekom dokaza se bomo sklicali na izreke \ref{izr:Hilbertopinv}, \ref{izr:Hilbertinvset} in \ref{izr:Liouville}. Naj bo $X$ poljuben kompleksen Hilbertov prostor in vzemimo poljuben operator $T\in B(X)$.
	\begin{enumerate}[a)]
		\item Vzemimo poljuben element spektra $\lambda\in\sigma(T)$ in denimo, da je $\norm{T} < \abs{\lambda}$. Tedaj je $\norm{\lambda^{-1}T} < 1$ in po izreku \ref{izr:Hilbertopinv} sledi, da je operator $I - \lambda^{-1}T$ obrnljiv. To pomeni, da je tudi operator $-\lambda(I - \lambda^{-1}T)= T - \lambda I$ obrnljiv. Sledi, da $\lambda\notin\sigma(T)$. Posledično, če je $\lambda\in \sigma(T)$, nujno velja, da je $\abs{\lambda}\leq \norm{T}$.
		
		\item Opazimo, da smo v točki $a)$ hkrati tudi pokazali, da je spekter omejena podmnožica v $\C$. Zato za dokaz te točke zadošča, da pokažemo, da je tudi zaprta podmnožica v $\C$. Definiramo preslikavo $\map{f}{\C}{B(X)}$ s predpisom $f(\nu) = T - \nu I$ in opazimo: $$\norm{f(\mu) - f(\lambda)} = \abs{\mu - \lambda}\cdot\norm{I}$$ za vsaka $\mu, \lambda \in \C$.
		Vzemimo sedaj poljubno število $\varepsilon>0$ in izberimo $\delta = \frac{\varepsilon}{\norm{I}}$. Če je $\abs{\mu - \lambda}<\delta$, je: $$\norm{f(\mu) - f(\lambda)} < \frac{\varepsilon}{\norm{I}}\norm{I} = \varepsilon.$$ To pomeni, da je preslikava $f$ enakomerno zvezna. 
		Sedaj z $\mathcal{G}$ označimo množico vseh obrnljivih omejenih linearnih operatorjev na $X$. Vemo že, da je $\lambda\in\sigma(T)$ natanko tedaj, ko je $T-\lambda I \in \mathcal{G}^{c}$. Slednje je ekvivalnento temu, da je $f(\lambda)\in\mathcal{G}^{c}$, to bo pa res natanko tedaj, ko bo $\lambda$ element praslike $f^{-1}(\mathcal{G}^{c})$.
		Posledično je $f^{-1}(\mathcal{G}^{c}) = \sigma(T)$. Izrek \ref{izr:Hilbertinvset} nam pove, da je množica $\mathcal{G}$ v $B(X)$ odprta, torej je $\mathcal{G}^{c}$ v $B(X)$ zaprta. Ker je preslikava $f$ zvezna, je množica $f^{-1}(\mathcal{G}^{c})$ zaprta v $\C$, torej je tudi $\sigma(T)$ zaprta v $\C$. Po izreku \ref{izr:HBL} je potem $\sigma(T)$ kompaktna podmnožica v $\C$.
		
		\item Dokaz bomo izvedli s protislovjem. Denimo, da je $\sigma(T)=\emptyset$. To pomeni, da je za vsak skalar $\lambda\in\C$ operator $T - \lambda I$ obrnljiv. Označimo njegov inverz s $(T - \lambda I)^{-1}$. Naj bo $\map{f}{B(X)}{\C}$ poljuben omejen (oz. zvezen) linearen funkcional. Definiramo funkcijo $\map{\varphi}{\C}{\C}$ s predpisom $\varphi(\lambda)= f((T-\lambda I)^{-1})$. Zaradi naše začetne predpostavke je funkcija $\varphi$ dobro definirana in zvezna. Premislimo, da je funkcija $\varphi$ kompleksno odvedljiva na $\C$ in omejena.
		Najprej bomo premislili odvedljivost. Vzemimo poljubno kompleksno število $\lambda_0\in\C$. Tedaj je: \begin{align*}
			\frac{\varphi(\lambda)-\varphi(\lambda_0)}{\lambda - \lambda_0} &= \frac{1}{\lambda - \lambda_0}(f((T-\lambda I)^{-1}) - f((T-\lambda_0 I)^{-1})) \\
			&= \frac{1}{\lambda - \lambda_0}f((T-\lambda I)^{-1} - (T-\lambda_0 I)^{-1}) \\
			&= \frac{1}{\lambda - \lambda_0}f((T-\lambda_0 I)^{-1}\left((T-\lambda_0 I) - (T-\lambda I)\right)(T-\lambda I)^{-1}) \\
			&= \frac{1}{\lambda - \lambda_0}f((\lambda - \lambda_0)(T-\lambda_0 I)^{-1}(T-\lambda I)^{-1}) \\
			&= f((T-\lambda_0 I)^{-1}(T-\lambda I)^{-1}).
		\end{align*}
		Posledično je: \begin{align*}
			\varphi'(\lambda_0)=\lim_{\lambda\to \lambda_0}\frac{\varphi(\lambda)-\varphi(\lambda_0)}{\lambda - \lambda_0} &= \lim_{\lambda\to\lambda_0}f((T-\lambda_0 I)^{-1}(T-\lambda I)^{-1}) \\
			&= f(\lim_{\lambda\to \lambda_0}\left((T-\lambda_0 I)^{-1}(T-\lambda I)^{-1}\right)) \\
			&= f((T-\lambda_0 I)^{-1}\lim_{\lambda\to \lambda_0}(T-\lambda I)^{-1}) \\
			&= f(((T-\lambda_0 I)^{-1})^2).
		\end{align*}
		Ker ta limita obstaja za vsak $\lambda_0\in\C$, je funkcija $\varphi$ kompleksno odvedljiva na celotni kompleksni ravnini $\C$. Premislimo še, da je $\varphi$ omejena.
		Vzemimo najprej poljubno kompleksno število $\lambda\in\C$, za katerega je $\abs{\lambda}>\norm{T}$. Tedaj je: $$(T- \lambda I)^{-1} = (-\lambda)^{-1}(I - \lambda^{-1}T)^{-1}.$$ Opazimo, da je $\norm{\lambda^{-1}T}<1$. Izrek \ref{izr:Hilbertopinv} nam pove, da je potem: $$(T- \lambda I)^{-1} = (-\lambda)^{-1}\sum_{n = 1}^{\infty}(\lambda^{-1}T)^n.$$ Posledično je: $$\norm{(T- \lambda I)^{-1}} \leq\frac{1}{\abs{\lambda}}\sum_{n = 1}^{\infty}\abs{\lambda}^{-n}\norm{T}^n = \frac{1}{\abs{\lambda}}\sum_{n=1}^{\infty}\left(\frac{\norm{T}}{\abs{\lambda}}\right)^n = \frac{1}{\abs{\lambda}}\frac{1}{1-\frac{\norm{T}}{\abs{\lambda}}}=\frac{1}{\abs{\lambda}-\norm{T}}.$$
		Ker je $0 \leq \norm{(T- \lambda I)^{-1}}$ za vsako število $\lambda\in\C$ in ker je $$\lim_{\abs{\lambda}\to\infty}\frac{1}{\abs{\lambda} - \norm{T}}=0$$ po t.~i.~izreku o sendviču sklepamo, da je: $$\lim_{\abs{\lambda}\to\infty}\norm{(T- \lambda I)^{-1}} = 0.$$ Z dobljeno limito bomo obravnavali obnašanje funkcije $\varphi$ daleč od izhodišča. Za vsako kompleksno število $\nu\in\C$ velja naslednja ocena: $$0 \leq\abs{\varphi(\nu)} = \abs{f((T-\nu I)^{-1})} \leq \norm{f}\cdot\norm{(T-\nu I)^{-1}}.$$ Z uporabo prej izračunane limite in ponovno uporabo izreka o sendviču vidimo, da je $\lim_{\abs{\lambda}\to\infty}\abs{\varphi(\lambda)}=0$. Posledično je: $$\lim_{\abs{\lambda}\to\infty}\varphi(\lambda)=0.$$ To pomeni, da obstaja tako realno število $R>0$, da je $\abs{\varphi(\lambda)}\leq 1$ za vsako število $\lambda\in \overline{K_R(0)}^c$. Ker je $\varphi$ zvezna na $\overline{K_R(0)}$, je tam tudi omejena. To pomeni, da je $\varphi$ omejena na celi kompleksni ravnini $\C$.
		Izrek \ref{izr:Liouville} nam pove, da je potem funkcija $\varphi$ konstantna. Ker je, $\lim_{\abs{\lambda}\to\infty}\varphi(\lambda) = 0$ in je funkcija $\varphi$ zvezna, je potem $\varphi\equiv 0$. To pomeni, da je $f((T - \lambda I)^{-1})=0$ za vsak $\lambda\in\C$. Ker je $f$ bil poljuben omejen linearen funkcional na $B(X)$, sklep velja za vse omejene linearne funkcionale na $B(X)$. Vzemimo sedaj poljubno kompleksno število $\lambda\in\C$.
		Posledica \ref{posl:HB} nam pove, da obstaja tak omejen linearen funkcional $\map{g}{B(X)}{\C}$, da je $0 = g((T- \lambda I)^{-1})=\norm{(T- \lambda I)^{-1}}$. To pa pomeni, da je $(T- \lambda I)^{-1} = 0$. Prišli smo v protislovje, saj ničelni operator ni obrnljiv.
	\end{enumerate}
\end{dokaz}

\begin{opomba}
	\label{opomb:specprimeri} Naj bo $X$ poljuben kompleksen Hilbertov prostor. Prva točka izreka \ref{izr:specter} nam pove, da za poljuben $T\in B(X)$ velja, da je $\sigma(T)\subseteq \overline{K_{\norm{T}}(0)}$. S pomočjo druge točke izreka tudi opazimo dva skrajna primera za spekter. V prvem skrajnem primeru spekter vsebuje le eno samo vrednost: $\sigma(T)=\{\lambda_0\}$, za nek $\lambda_0\in\overline{K_{\norm{T}}(0)}$. Primer takega operatorja $T$ je ravno ničelni operator: $\sigma(0) = \{0\}$. Drugi skrajni primer je, da je $\sigma(T) = \overline{K_{\norm{T}}(0)}$.
\end{opomba}

Sedaj premislimo še povezavo med spektrom poljubnega operatorja in spektrom njegovega adjungiranega operatorja.

\begin{lema}
	\label{lem:adjspec}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor. Tedaj je za vsak $T\in B(x)$: $$\sigma(T^*)=\{\bar{\lambda}~|~\lambda\in\sigma(T)\}.$$
\end{lema}

\begin{dokaz}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor in denimo, da obstaja tak $\lambda\in\C$, ki ni v spektru operatorja $T\in B(X)$. Tedaj je operator $T-\lambda I$ obrnljiv in po lemi \ref{lem:adjinv} je potem tudi operator $(T-\lambda I)^* = T^* - \bar{\lambda}I$ obrnljiv. Sledi, da število $\bar{\lambda}$ ne pripada $\sigma(T^*)$. Na enak način premislimo, da če število $\bar{\lambda}$ ni element $\sigma(T^*)$, potem tudi $\lambda$ ne pripada $\sigma(T)$. To pomeni, da je $\sigma(T^*) =\{\bar{\lambda}~|~\lambda\in\sigma(T)\}$.
\end{dokaz}
Poleg ravnokar pokazane zveze, za spekter operatorja veljata še dve pomembni zvezi, ki ju bomo navedli in dokazali v naslednjem izreku.
\begin{izrek}
	\label{izr:polyspec}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor ter naj bo $T\in B(X)$ poljuben operator. Tedaj za poljuben nekonstanten polinom $p\in\C[X]$ velja: \begin{enumerate}[a)]
		\item $\sigma(p(T))=\{p(z)~|~z\in\sigma(T)\}$,
		\item Če je $T$ obrnljiv, je $\sigma(T^{-1})=\{\lambda^{-1}~|~\lambda\in\sigma(T)\}$.
	\end{enumerate}
\end{izrek}

\begin{dokaz}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor in $T\in B(X)$ poljuben operator.
	\begin{enumerate}[a)]
		\item Vzemimo poljuben nekonstanten polinom $p\in\C[X]$ in poljubno število $\lambda\in\C$. Definiramo polinom $q$ s predpisom $q(z) = \lambda - p(z)$. Po osnovnem izreku algebre lahko polinom $q$ zapišemo kot produkt linearnih faktorjev: $q(z) = a(z-\lambda_1)(z-\lambda_2)\ldots(z-\lambda_n)$, kjer so $\lambda_1, \lambda_2,\ldots, \lambda_n\in\C$ (ne nujno različne) ničle polinoma in $a\in\C\setminus\{0\}$.
		Velja: \begin{align*}
			\lambda\notin \sigma(p(T)) &\iff~\text{Operator}~\lambda I - p(T)~ \text{je obrnljiv.} \\
			&\iff q(T)=a(T-\lambda_1 I)(T-\lambda_2 I)\ldots(T-\lambda_n I)~\text{je obrnljiv.} \\
			&\iff~\text{Operator}~(T-\lambda_i I)~\text{je obrnljiv za vsak }~i\in\{1, 2, \ldots, n\}. \\
			&\iff~\text{nobena ničla polinoma}~q~\text{ni vsebovana v}~\sigma(T). \\
			&\iff q(z)\neq 0~\text{za vsak}~z\in\sigma(T). \\
			&\iff \lambda\neq p(z)~\text{za vsak}~z\in\sigma(T).			
		\end{align*}
		Če povzamemo: Število $\lambda$ ni vsebovano v spektru $\sigma(p(T))$ natanko tedaj, ko je $p(z)\neq \lambda$ za vsak $z\in\sigma(T)$. Ko to negiramo, dobimo naslednjo ekvivalenco: $$\lambda\in\sigma(p(T))~\text{natanko tedaj, ko obstaja tak}~z\in\sigma(T),~\text{da je}~p(z) = \lambda.$$ To pomeni, da je $\sigma(p(T))=\{p(z)~|~z\in\sigma(T)\}$.
		\item Predpostavimo, da je operator $T$ obrnljiv. Posledično $0$ ni vsebovana niti v $\sigma(T)$ niti v $\sigma(T^{-1})$ in zato lahko vsak element $\sigma(T^{-1})$ zapišemo kot inverz nekega $\mu\in\C$. Opazimo da je $\mu^{-1}I - T^{-1} = -\mu^{-1}T^{-1}(\mu I - T)$ in da je operator $\mu^{-1}T^{-1}$ obrnljiv. Sledi: \begin{align*}
			\mu^{-1}\in\sigma(T^{-1}) &\iff~\text{Operator}~\mu^{-1}I - T^{-1}~\text{ni obrnljiv.} \\
			&\iff~\text{Operator}~-\mu^{-1}T^{-1}(\mu I - T)~\text{ni obrnljiv.}\\
			&\iff~\text{Operator}~(\mu I - T)~\text{ni obrnljiv.} \\
			&\iff \mu\in\sigma(T).
		\end{align*}
		To pomeni, da je $\sigma(T^{-1})=\{\mu^{-1}~|~\mu\in\sigma(T)\}$.
	\end{enumerate}
\end{dokaz}

\section[Spektri posebnih tipov operatorjev]{Spektri posebnih tipov operatorjev}

S pomočjo izreka \ref{izr:polyspec} lahko enostavno določimo spektre raznih posebnih tipov operatorjev. Definirajmo še dva izmed teh posebnih tipov.

\begin{definicija}
	\label{def:idemnilop}
	Naj bosta $U$ in $V$ poljubna vektorska prostora nad poljem $\F$. Za linearen operator $\map{A}{U}{V}$ pravimo, da je: \begin{enumerate}[a)]
		\item \pojem{Idempotent}, če je $A^2 = A$. Operatorja $0$ in $I$ imenujemo \pojem{trivialna idempotenta}.
		\item \pojem{Nilpotent}, če obstaja tako število $n\in\N$, da je $A^n = 0$.
	\end{enumerate}
\end{definicija}
V naslednji lemi bomo premislili posebne lastnosti spektrov nilpotentnih, idempotentnih in unitarnih operatorjev.
\begin{lema}
	\label{lem:idenilunispec}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor. Za operator $T\in B(X)$ velja: \begin{enumerate}[a)]
		\item Če je $T$ netrivialen idempotent, je $\sigma(T) = \{0, 1\}$.
		\item Če je $T$ nilpotent, je $\sigma(T) = \{0\}$.
		\item Če je $T\in \mathcal{U}(X)$, je $\sigma(T) \subseteq \{\lambda\in\C~|~ \abs{\lambda}=1\}$.
	\end{enumerate}
\end{lema}

\begin{dokaz}
	Ključni del dokaza vseh trditev bo uporaba izreka \ref{izr:polyspec}. Naj bo $X$ poljuben kompleksen Hilbertov prostor in vzemimo poljuben operator $T\in B(X)$. \begin{enumerate}[a)]
		\item Denimo, da je operator $T\in B(X)\setminus\{0, I\}$ idempotent. Potem operator $T$ zadošča enačbi $T^2 - T = 0$. Označimo polinom $p(z) = z^2 - z$. Tedaj operator $T$ zadošča enačbi $p(T) = 0$. Kot smo že omenili v opombi \ref{opomb:specprimeri}, je $\sigma(0) = \{0\}$. Prva točka izreka \ref{izr:polyspec} nam pove: $$\{0\}=\sigma(0)=\sigma(p(T))=\sigma(T^2 - T) = \{\lambda^2 - \lambda~|~\lambda\in\sigma(T)\}.$$ Za vsak $\lambda\in\sigma(T)$ torej velja, da je $\lambda^2 - \lambda = 0$. Ta enačba ima dve rešitvi: $\lambda=0$ in $\lambda=1$. To pomeni, da je $\sigma(T)\subseteq\{0, 1\}$. Hkrati nam tretja točka izreka \ref{izr:specter} pove, da je $\sigma(T)\neq\emptyset$. Premislimo sedaj, da je $\{0, 1\}\subseteq\sigma(T)$.
		
		Denimo najprej, da $1\notin\sigma(T)$. Tedaj je operator $T- I$ obrnljiv, torej lahko enačbo $$(T-I)T = P(T) = 0$$ z leve pomnožimo z inverznim operatorjem $(T-I)^{-1}$. Če to storimo, dobimo enakost $T = 0$. To nas privede v protislovje z začetno predpostavko, da je $T\in B(X)\setminus\{0, I\}$. Sledi, da je $1\in\sigma(T)$.
		
		Denimo sedaj, da $0\notin\sigma(T)$. Tedaj je operator $T$ obrnljiv, torej lahko enačbo $$(T-I)T = P(T) = 0$$ z desne pomnožimo z inverznim operatorjem $T^{-1}$. Tako dobimo enakost $T-I = 0$ oziroma $T=I$. Tudi to nas privede v protislovje z enako začetno predpostavko, kot v prejšnji točki. To pomeni, da je $0\in\sigma(T)$. Sledi, da je $\sigma(T) = \{0, 1\}$.
		
		\item Naj bo sedaj operator $T\in B(X)$ nilpotent in naj bo $n$ najmanjše tako naravno število, za katerega je $T^n = 0$. Označimo polinom $q(z)=z^n$. Tedaj operator $T$ zadošča enačbi $q(T) = 0$. Po prvi točki izreka \ref{izr:polyspec} je potem: $$\{0\}=\sigma(0)=\sigma(q(T))=\sigma(T^n)=\{\lambda^n~|~\lambda\in\sigma(T)\}.$$ Sledi, da za vsak $\lambda\in\sigma(T)$ velja, da je $\lambda^n = 0$. Ta enačba ima le eno rešitev $\lambda=0$. To pomeni, da je $\sigma(T)\subseteq\{0\}$. Ker po tretji točki izreka \ref{izr:specter} spekter ne more biti prazna množica, sledi, da je $\sigma(T)=\{0\}$.
		
		\item Naj bo sedaj $T\in \mathcal{U}(X)$. Prva točka leme \ref{lem:unitarset} nam pove, da je $\norm{T}=1$. Posledično je: $$\sigma(T)\subseteq \overline{K_1(0)} = \{\lambda\in\C~|~\abs{\lambda}\leq 1\}.$$ Istočasno opazimo, da je po enakem argumentu tudi $\sigma(T^*) \subseteq \overline{K_1(0)}$. Sedaj upoštevamo, da je $T$ unitaren operator, torej je $T^* = T^{-1}$. Po drugi točki izreka \ref{izr:polyspec} potem velja: $$\sigma(T) =
		\{\lambda^{-1}~|~\lambda\in\sigma(T^*)\}\subseteq \{\lambda\in\C~|~\abs{\lambda}\geq 1\}.$$ Ker je $\sigma(T)$ hkrati podmnožica $\overline{K_1(0)}$ in $K_1^c(0)$, je vsebovana tudi v njunem preseku $\overline{K_1(0)}\cap K_1^c(0)$. Ta je ravno rob enotske krogle $K_1(0)$. To pomeni, da je: $$\sigma(T)\subseteq\{\lambda\in\C~|~\abs{\lambda}=1\}.$$
	\end{enumerate}
\end{dokaz}

\begin{opomba}
	\label{opomb:ortprojspec}
	Ker so ortogonalni projektorji posebni tipi idempotentov, nam lema \ref{lem:idenilunispec} pove, da je spekter vsakega ortogonalnega projektorja na kompleksnem Hilbertovem prostoru natanko množica $\{0, 1\}$.
\end{opomba}

Pri obravnavi spektrov normalnih in sebi-adjungiranih operatorjev nam bosta prišla prav naslednja nova pojma.

\begin{definicija}
	\label{def:specrad}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor in naj bo $T\in B(X)$ poljuben operator. \begin{enumerate}[a)]
		\item \pojem{Spektralni radij} operatorja $T$, označen z $r_\sigma(T)$, je definiran na naslednji način: $$r_\sigma(T) = \sup\{\abs{\lambda}~|~\lambda\in\sigma(T)\}.$$
		\item \pojem{Numerični razpon} operatorja $T$, označen z $V(T)$, je definiran na naslednji način: $$V(T)=\{\Sp{Tx}{x}~|~x\in X \land \norm{x}=1\}.$$
	\end{enumerate}
\end{definicija}

V primeru normalnih operatorjev velja naslednja zveza med spektrom in numeričnim razponom.

\begin{lema}
	\label{lem:normalnumrazspec}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor. Če je operator $T\in B(X)$ normalen, je $\sigma(T)\subseteq \overline{V(T)}$.
\end{lema}

\begin{dokaz}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor in $T\in B(X)$ poljuben normalen operator. Vzemimo poljubno število $\lambda\in\sigma(T)$. V primeru \ref{prim:Tminlamnorm} smo že premislili, da je operator $(T-\lambda I)$ normalen. Ker operator $(T - \lambda I)$ ni obrnljiv, po posledici \ref{posl:normalinv} za vsak $r>0$ obstaja nek element $x_r\in X$, taki, da je $$\norm{(T-\lambda I)x_r}< r\norm{x_r}.$$ Posledično v $X$ obstaja tako zaporedje $\{x_n\}_{n\in\N}$, da je $\norm{x_n} = 1$ za vse $n\in\N$ in za katero je: $$\lim_{n\to\infty}\norm{(T-\lambda I)x_n} = 0.$$
	Po izreku \ref{izr:CSB} za vsako število $n\in\N$ velja: $$0\leq\abs{\Sp{(T-\lambda I)x_n}{x_n}} \leq \norm{(T-\lambda I)x_n}\cdot\norm{x_n} = \norm{(T-\lambda I)x_n}.$$ Izrek o sendviču nam sedaj pove, da je $\lim_{n\to\infty}\Sp{(T-\lambda I)x_n}{x_n} = 0$. Slednjo enakost razpišemo v: $$\lim_{n\to\infty}(\Sp{Tx_n}{x_n}-\lambda\Sp{x_n}{x_n})=0.$$ Ko upoštevamo, da je $\Sp{x_n}{x_n} = 1$ za vsak $n\in\N$, dobimo, da je: $$\lim_{n\to\infty}\Sp{Tx_n}{x_n} = \lambda.$$ Sledi, da je $\lambda \in \overline{V(T)}$.
\end{dokaz}

V naslednjem izreku so opisane lastnosti spektrov in numeričnih razponov sebi-adjungiranih operatorjev.

\begin{izrek}
	\label{izr:sebiadjnumraz}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor in naj bo $T\in B(X)$ poljuben sebi-adjungiran operator. Velja: \begin{enumerate}[a)]
		\item $V(T)\subseteq \R$,
		\item $\sigma(T)\subseteq \R$,
		\item $\norm{T}\in\sigma(T)\lor (-\norm{T})\in\sigma(T)$,
		\item $r_\sigma(T)=\sup\{\abs{t}~|~t\in V(T)\} = \norm{T}$,
		\item $\inf\{\lambda~|~\lambda\in\sigma(T)\} \leq t \leq \sup\{\lambda~|~\lambda\in\sigma(T)\}$ za vsak $t\in V(T)$.
	\end{enumerate}
\end{izrek}

\begin{dokaz}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor in naj bo $T\in B(X)$ poljuben sebi-adjungiran operator.
	\begin{enumerate}[a)]
		\item Ta trditev je direktna posledica trditve \ref{trd:sebiadjop}.
		\item Po prejšnji točki vemo, da je $V(T)\subseteq \R$. Posledično je tudi $\overline{V(T)}\subseteq\R$. Spomnimo se opombe \ref{opomb:sebiadjsonormal}, ki nam pove, da je vsak sebi-adjungiran operator tudi normalen. Na koncu nam lema \ref{lem:normalnumrazspec} pove, da je $\sigma(T)\subseteq\overline{V(T)}\subseteq \R$.
		\item Če je $T=0$, je po opombi \ref{opomb:specprimeri} $\sigma(T) = \sigma(0)=\{0\}$. V tem primeru trditev očitno velja. Denimo torej, da je $T\neq 0$ in označimo $\norm{T} = a$. Pokazali bomo, da je $a^2\in\sigma(T^2)$. Po definiciji norme operatorja v $X$ obstaja tako zaporedje $\{x_n\}_{n\in\N}$, da je $\norm{x_n}=1$ za vsak $n\in\N$, za katerega je $\lim_{n\to\infty}\norm{Tx_n} = \norm{T} = a$. Posledično za vsak $n\in\N$ velja: \begin{align*}
			\norm{(a^2I - T^2)x_n}^2 &= \Sp{(a^2I - T^2)x_n}{(a^2I - T^2)x_n} \\
			&= \Sp{a^2x_n}{a^2x_n} - \Sp{T^2x_n}{a^2x_n} - \Sp{a^2x_n}{T^2x_n} + \Sp{T^2x_n}{T^2x_n} \\
			&= \norm{a^2x_n}^2 - a^2\Sp{T^2x_n}{x_n} - a^2\Sp{x_n}{T^2x_n} + \norm{T^2x_n}^2.
		\end{align*}
		Ker je $T$ sebi-adjungiran je $\Sp{T^2x}{x} = \Sp{x}{T^2x}$za vsak $x\in X$. To upoštevamo v našem računu in tako za vsak $n\in\N$ dobimo: 
		\begin{align*}
			\norm{(a^2I - T^2)x_n}^2 &= \norm{a^2x_n}^2 - 2a^2\Sp{T^2x_n}{x_n} + \norm{T^2x_n}^2 \\
			&\leq a^4\norm{x_n}^2 + \norm{T^2}^2\norm{x_n}^2 - 2a^2\Sp{T^2x_n}{x_n}  \\
			&\leq a^4 + a^4\norm{x_n}^2 - 2a^2\Sp{Tx_n}{Tx_n}  \\
			&= 2a^4 - 2a^2\norm{Tx_n}^2 = 2a^2(a^2 - \norm{Tx_n}^2).
		\end{align*}
		Ko upoštevamo, da je $\lim_{n\to\infty}\norm{Tx_n} = a$, ter da je kvadriranje zvezna preslikava, vidimo, da je je $\lim_{n\to\infty}\norm{Tx_n}^2 = a^2$. To pomeni, da je: $$\lim_{n\to\infty}\norm{(a^2I - T^2)x_n}^2 = 0.$$ Sledi, da za vsako število $r>0$ obstaja tak $n\in \N$, da je $\norm{(a^2I-T^2)x_n} < r$. Zato po posledici \ref{posl:normalinv} operator $a^2I - T^2$ ni obrnljiv. To pomeni, da je $a^2 \in \sigma(T^2)$. 
		Prva točka izreka \ref{izr:polyspec} nam pove, da je $a^2\in \{\lambda^2~|~\lambda\in \sigma(T)\}$. Posledično je vsaj eno izmed števil $a=\norm{T}$ in $-a$ vsebovano v $\sigma(T)$.
		\item Upoštevajoč prejšnjo točko in definicijo spektralnega radija vidimo, da je $\norm{T}\leq r_\sigma(T)$. Ker je po lemi \ref{lem:normalnumrazspec} ~$\sigma(T) \subseteq \overline{V(T)}$, velja: $$r_\sigma(T) \leq \sup\{\abs{\lambda}~|~\lambda\in V(T)\}.$$ Vsak element množice $V(T)$ je po definiciji oblike $\Sp{Tx}{x}$, za nek $x\in X$ za katerega je $\norm{x}=1$. Po izreku \ref{izr:CSB} potem za vsak element $\abs{\Sp{Tx}{x}}\in \{\abs{\lambda}~|~\lambda\in V(T)\}$ velja: $$\abs{\Sp{Tx}{x}}\leq \norm{Tx}\cdot\norm{x} = \norm{Tx}\leq \norm{T}.$$ Posledično je tudi supremum te množice navzgor omejen z $\norm{T}$. Velja torej: $$\norm{T} \leq r_\sigma(T) \leq \sup\{\abs{\lambda}~|~\lambda\in V(T)\} \leq \norm{T}.$$ Od tod sledi, da je $r_\sigma(T)=\sup\{\abs{t}~|~t\in V(T)\} = \norm{T}$.
		\item Vpeljimo oznaki $\alpha=\inf\{\lambda~|~\lambda\in\sigma(T)\}$ in $\beta=\sup\{\lambda~|~ \lambda\in\sigma(T)\}$. Vzemimo sedaj poljubno število $t\in V(T)$ in naj bo $x\in X$ tak, da je $\norm{x}=1$ in $\Sp{Tx}{x}=t$. 
		Denimo najprej, da je $t < \alpha$. Tedaj ima operator $\beta I - T$ po prvi točki izreka \ref{izr:polyspec} spekter $$\sigma(\beta I - T)=\{\beta - \lambda~|~\lambda\in\sigma(T)\},$$ ki leži znotraj intervala $[0, \beta - \alpha]$. Sledi, da je $r_\sigma(\beta I - T) \leq \beta - \alpha$. Ker je $\beta\in\R$ opazimo tudi, da je: $$(\beta I - T)^* = \bar{\beta}I - T^* = \beta I - T.$$ To pomeni, da je operator $\beta I - T$ sebi-adjungiran. Dodatno, ker je $t<\alpha$, je: $$\Sp{(\beta I - T)x}{x} = \beta\Sp{x}{x} - \Sp{Tx}{x} = \beta - t > \beta - \alpha.$$ Toda po točki $d)$ tega izreka je $r_\sigma(\beta I - T) = \sup\{\abs{\lambda}~|~\lambda\in V(\beta I - T)\}$. Zato za operator $\beta I - T$ velja: $$\beta - \alpha \geq r_\sigma(\beta I - T) = \sup\{\abs{\lambda}~|~\lambda\in V(\beta I - T)\}\geq\Sp{(\beta I - T)x}{x}>\beta - \alpha.$$ Prišli smo v protislovje. Sledi, da je $\alpha \leq t$.
		
		Denimo sedaj, da je $\beta < t$. Tedaj opazimo, da ima operator $T - \alpha I$ po prvi točki izreka \ref{izr:polyspec} spekter $$\sigma(T - \alpha I)=\{\lambda - \alpha~|~ \lambda \in \sigma(T)\},$$ ki leži znotraj intervala $[0, \beta - \alpha]$. Posledično velja, da je $r_\sigma(T - \alpha I)\leq \beta - \alpha$. Ker je $\alpha\in\R$ opazimo, na enak način kot za operator $\beta I - T$, da je tudi $T - \alpha I$ sebi-adjungiran operator. Dodatno nam točka $d)$ tega izreka pove, da je: $$r_\sigma(T - \alpha I) = \sup\{\abs{\lambda}~|~\lambda\in V(T - \alpha I)\}.$$ Ker je $\beta < t$ zato velja: $$\Sp{(T - \alpha I)x}{x} = \Sp{Tx}{x} - \alpha\Sp{x}{x} = t - \alpha > \beta - \alpha.$$
		To pomeni, da je: $$\beta - \alpha \geq r_\sigma(T - \alpha I) =\sup\{\abs{\lambda}~|~\lambda\in V(T - \alpha I)\}\geq \Sp{(T - \alpha I)x}{x} > \beta - \alpha.$$
		Ponovno smo prišli v protislovje. Sledi, da je $t \leq \beta$. Ko to ugotovitev združimo s prejšnjo ugotovitvijo o $\alpha$, dobimo: $$\alpha \leq t \leq \beta.$$ Ker je bil izbrani $t\in V(T)$ poljuben, ta sklep velja za vse elemente $V(T)$.
	\end{enumerate}
\end{dokaz}

S pomočjo pojma numeričnega razpona operatorja lahko sedaj obravnavamo še en poseben tip sebi-adjungiranih operatorjev.

\subsection{Koren pozitivno semidefinitnih operatorjev}

Na koncu poglavja o omejenih linearnih operatorjih smo definirali t.~i.~">koren operatorja"< in povedali, da za nadaljni študij potrebujemo znanje o spektru. Sedaj, ko smo zadostili omenjenemu pogoju, se lahko lotimo obravnave tega vprašanja na posebnem tipu sebi-adjungiranih operatorjev. Podajmo najprej definicijo le teh.

\begin{definicija}
	\label{def:pozop}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor. Pravimo, da je operator $T\in B(X)$ \pojem{pozitiven} operator, če je sebi-adjungiran in pozitivno semidefiniten. Za vsak $x\in X$ torej velja: $$0 \leq \Sp{Tx}{x}.$$
\end{definicija}

S pomočjo izreka \ref{izr:sebiadjnumraz} podamo karakterizacijo pozitivnih operatorjev.

\begin{trditev}
	\label{trd:pozopkar}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor in naj bo $T\in B(X)$ poljuben sebi-adjungiran operator. Operator je $T$ pozitiven natanko tedaj, ko je $\sigma(T)\subseteq [0, \infty)$.
\end{trditev}

\begin{dokaz}
	Dokazali bomo obe implikaciji. 
	Naj bo $X$ poljuben kompleksen Hilbertov prostor in naj bo $T\in B(X)$ poljuben sebi-adjungiran operator. Denimo najprej, da je operator $T$ pozitiven. Tedaj za vsak element $x\in X$ velja $0\leq\Sp{Tx}{x}$. To pomeni, da je $V(T)\subseteq [0, \infty)$. Ko upoštevamo lemo \ref{lem:normalnumrazspec}, opazimo naslednje: $$\sigma(T)\subseteq\overline{V(T)}\subseteq [0, \infty).$$ S tem smo dokazali prvo implikacijo.
	
	Denimo sedaj, da je $\sigma(T)\subseteq [0, \infty)$. Uporabili bomo točki $a)$ in $e)$ izreka \ref{izr:sebiadjnumraz}, ki nam poveta, da je $V(T)\subseteq\R$ in $\inf\{\lambda~|~\lambda\in\sigma(T)\} \leq t$ za vsak $t\in V(T)$. Ker je $\sigma(T)\subseteq [0, \infty)$ zato sledi, da je $0 \leq t$ za vsak $t\in V(T)$. To pomeni, da je $0\leq\Sp{Tx}{x}$ za vsak $x\in X$, torej je $T$ pozitiven operator.
\end{dokaz}

Navedimo sedaj nekaj primerov pozitivnih operatorjev.

\begin{primer}
	\label{prim:pozoptriv}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor. Preverimo, da sta $0$ in $I$ pozitivna operatorja. Vzemimo poljuben element $x\in X$. Tedaj je $$\Sp{0x}{x} = \Sp{0}{x} = 0 \geq 0$$ in $$\Sp{Ix}{x} = \Sp{x}{x} = \norm{x}^2 \geq 0.$$ To pomeni, da sta oba operatorja res pozitivna.
\end{primer}

\begin{primer}
	\label{prim:pozoptriv}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor in $T\in B(X)$ poljuben sebi-adjungiran operator. Dodatno, naj bosta $U, V\in B(X)$ poljubna pozitivna operatorja in $\lambda\in \C$ poljuben skalar. Preverimo, da so operatorji $T^*T$, $U+V$ in $\lambda U$ pozitivni operatorji. 
	V lemi \ref{lem:sebiadjlast} smo že premislili, da so operatorji $T^*T$, $U+V$ in $\lambda U$ sebi-adjungirani. Vzemimo poljuben element $x\in X$. 
	Za operator $T^*T$ tedaj velja: $$\Sp{T^*Tx}{x} = \Sp{Tx}{Tx} = \norm{Tx}^2 \geq 0.$$ To pomeni, da je $T^*T$ res pozitiven operator.
	Istočasno velja: $$\Sp{(U+V)x}{x} = \Sp{Ux}{x} + \Sp{Vx}{x} \geq 0$$
	in $$\Sp{\lambda Ux}{x} = \lambda\Sp{Ux}{x} \geq 0.$$
	Torej sta tudi $U+V$ in $\lambda U$ pozitivna operatorja.	
\end{primer}

Sedaj se vrnimo k vprašanju korena operatorja. Najprej bomo premislili en pomožni rezultat, s pomočjo katerega bomo opisali, kako lahko definiramo nepolinomske funkcije na operatorjih.

\begin{lema}
	\label{lem:sebiadjPhi}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor in označimo s $\mathcal{S}(X)$ realen Banachov prostor vseh sebi-adjungiranih operatorjev na $X$. Tedaj za vsak sebi-adjungiran operator $T\in \mathcal{S}(X)$ obstaja tak omejen linearen operator $\map{\Phi}{\mathcal{C}_\R(\sigma(T))}{\mathcal{S}(X)}$, da je: \begin{enumerate}
		\item $\Phi(p) = p(T)$ za vsak polinom $p\in\mathcal{C}_\R(\sigma(T))$,
		\item $\Phi(f\cdot g) = \Phi(f)\cdot\Phi(g)$ za vse $f, g\in \mathcal{C}_\R(\sigma(T))$.
	\end{enumerate}
\end{lema}

\begin{dokaz}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor in vzemimo poljuben sebi-adjungiran operator $T\in\mathcal{S}(X)$. \begin{enumerate}
		\item Označimo s $P$ vektorski podprostor vseh polinomov v prostoru $\mathcal{C}_\R(\sigma(T))$ in definiramo preslikavo $\map{\varphi}{P}{\mathcal{S}(X)}$ s predpisom $\varphi(p) = p(T)$. Preverimo najprej, da je $\varphi$ linearna transformacija. Vzemimo poljubna polinoma $p, q\in \mathcal{C}_\R(\sigma(T))$ ter poljubna skalarja $\alpha_1,\alpha_2\in\C$. Tedaj je: $$\varphi(\alpha_1p+\alpha_2q) =(\alpha_1p + \alpha_2q)(T) = \alpha_1p(T) + \alpha_2q(T) = \alpha_1\varphi(p) + \alpha_2\varphi(q).$$
		To pomeni, da je preslikava $\varphi$ linearna. Še več, za poljubna polinoma $p, q\in P$ je: $$\varphi(p\cdot q) = (p\cdot q)(T) = p(T)\cdot q(T) = \varphi(p)\cdot \varphi(q).$$ Naj bo sedaj $p\in P$ poljuben polinom. V naslednjem premisleku bomo uporabili točko $d)$ izreka \ref{izr:sebiadjnumraz} in izrek \ref{izr:polyspec}: \begin{align*}
			\norm{\varphi(p)} &= \norm{p(T)} = r_\sigma(p(T)) \\
			&= \sup\{\abs{\lambda}~|~\lambda\in\sigma(p(T))\} \\ &= \sup\{\abs{p(\lambda)}~|~\lambda\in\sigma(T)\} = \norm{p}.
		\end{align*}
		To pomeni, da je preslikava $\varphi$ linearna izometrija. Ker je podprostor $P$ gosta podmnožica prostora $\mathcal{C}_\R(\sigma(T))$ ter ker je $\mathcal{S}(X)$ realen Banachov prostor, po točki $b)$ izreka \ref{izr:opHB} sledi, da obstaja tak operator $\Phi\in B(\mathcal{C}_\R(\sigma(T)))$, za katerega je $\Phi(p) = \varphi(p)$ za vsak polinom $p\in P$. Naj bosta sedaj $f, g\in \mathcal{C}_\R(\sigma(T))$ poljubni funkciji. Ker je $P$ gosta podmnožica $\mathcal{C}_\R(\sigma(T))$, obstajata taki zaporedji $\{p_n\}_{n\in\N}$ in $\{q_n\}_{n\in\N}$, da je $\lim_{n\to\infty}p_n = f$ in $\lim_{n\to\infty}q_n = g$. Zato velja: \begin{align*}
			\Phi(f\cdot g) &= \Phi\left(\lim_{n\to \infty}p_n \cdot \lim_{n\to\infty}q_n\right) = \Phi\left(\lim_{n\to \infty}(p_n \cdot q_n)\right) \\ &= \lim_{n\to \infty}\Phi(p_n\cdot q_n) = \lim_{n\to \infty}(\Phi(p_n)\cdot \Phi(q_n)) \\ &= \lim_{n\to \infty}\Phi(p_n) \cdot \lim_{n\to \infty}\Phi(q_n) = \Phi\left(\lim_{n\to \infty}p_n\right) \cdot \Phi\left(\lim_{n\to \infty}q_n\right) \\ &= \Phi(f) \cdot \Phi(g).
		\end{align*}
	\end{enumerate}
\end{dokaz}
Lema \ref{lem:sebiadjPhi} nam zagotovi obstoj zvezne linearne preslikave, ki danemu opeatorju $T\in\mathcal{S}(X)$ in dani zvezni realni funkciji na $\sigma(T)$ določi operator $\Phi(f)$. Le tega pogosto označimo kar z oznako $f(T)$. S pomočjo tega rezultata lahko torej obravnavamo funkcije v operatorjih. V naslednjem izreku se bomo posebej zanimali za funkciji kvadratnega in četrtega korena na $[0, \infty)$.
\begin{izrek}
	\label{izr:pozroot}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor in označimo s $\mathcal{S}(X)$ realen Banachov prostor sebi-adjungiranih operatorjev na $X$. Tedaj za pozitiven operator $T\in\mathcal{S}(X)$ velja, da obstaja enolično določen pozitiven koren $U$ operatorja $T$, ki je limita nekega zaporedja polinomov v $T$.
\end{izrek}
\begin{dokaz}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor in vzemimo poljuben pozitiven operator $T\in\mathcal{S}(X)$. Najprej bomo premislili obstoj, nato pa enoličnost korena.
	Naj bo $P$ vektorski podprostor prostora $\mathcal{C}_\R(\sigma(T))$, ki vsebuje vse polinome. Trditvev \ref{trd:pozopkar} nam pove, da je $\sigma(T)\subseteq [0, \infty)$. Posledično lahko na $\sigma(T)$ dobro definiramo funkciji $\map{f, g, }{\sigma(T)}{\R}$ s predpisoma $f(t) = t^{\frac{1}{4}}$ in $g(t)=t^{\frac{1}{2}}$. Opazimo, da sta $f$ in $g$ zvezni realni funkciji na $\sigma(T)$. Vzemimo sedaj omejen linearen operator $\map{\Phi}{\mathcal{C}_\R(\sigma(T))}{\mathcal{S}(X)}$ iz leme \ref{lem:sebiadjPhi} in z njegovo pomočjo definiramo operatorja $U = \Phi(g)=g(T)$ in $V = \Phi(f)=f(T)$. Operatorja $V$ in $U$ sta seveda oba sebi-adjungirana. Še več, ker je prostor polinomov na $\sigma(T)$ gosta podmnožica v $\mathcal{C}_\R(\sigma(T))$, je funkcija $g$ limita nekega zaporedja polinomov $\{p_n\}_{n\in\N}$ iz $P$. Posledično je operator $U=g(T)$ limita zaporedja teh istih polinomov, izračunana v $T$: $$g(T)=\Phi(g)=\Phi\left(\lim_{n\to \infty}p_n\right)=\lim_{n\to \infty}\Phi(p_n)=\lim_{n\to \infty}p_n(T).$$ Hkrati velja tudi, da je: $$U^2 = (g(T))^2 = id(T) = T.$$ To pomeni, da je operator $U$ koren operatorja $T$. Premislimo še, da je operator $U$ pozitiven. Opazimo: $$V^2 = (f(T))^2 = g(T) = U.$$ Vzemimo sedaj poljuben element $x\in X$. Ker je operator $V$ sebi adjungiran velja naslednje: $$\Sp{Ux}{x} = \Sp{V^2x}{x} = \Sp{Vx}{Vx} = \norm{Vx}^2 \geq 0.$$
	Ker to velja za vsak element $x\in X$ sledi, da je operator $U$ res pozitiven.
	
	Denimo sedaj, da operator $T$ premore še en pozitiven koren, ki ga označimo z $Q$. Tedaj je $$QT = QQ^2 = Q^2Q = TQ.$$ Še več, hitro vidimo, da je za vsak $n\in\N$ tudi $QT^n = T^nQ$. Vzemimo sedaj poljuben polinom $p\in P$ s predpisom $p(x) = a_0 + a_1x + \ldots + a_nx^n$. Tedaj velja: $$Qp(T) = Q(a_0I + a_1T + \ldots + a_nT^n) = (a_0I + a_1T + \ldots + a_nT^n)Q = p(T)Q.$$ Ker je operator $U$ limita zaporedja $\{p_n(T)\}_{n\in\N}$, sledi, da je tudi $QU = UQ$. Ker je operator $Q$ pozitiven, po prejšnjem premisleku, premore pozitiven koren, ki ga označimo z $R$. Vsemimo sedaj poljuben element $x\in X$ in naj bo $y = (U-Q)x$. Ker je $U^2 = Q^2 = T$, sledi naslednje: \begin{align*}
		\norm{Vy}^2 + \norm{Ry}^2 &= \Sp{V^2y}{y} + \Sp{R^2y}{y} = \Sp{(U+Q)y}{y} \\
		&= \Sp{(U + Q)(U-Q)x}{y} = \Sp{(U^2 + QU - UQ - Q^2)x}{y} \\
		&= \Sp{(U^2 - Q^2)x}{y} = 0.
	\end{align*}
	To pomeni, da sta $\norm{Vy}=0$ in $\norm{Ry}=0$ oziroma $Vy = 0$ in $Ry = 0$. Posledično je tudi $Uy = V^2y = 0$ in $Qy = R^2y = 0$. Od tod sledi naslednje: $$\norm{(U-Q)x}^2=\Sp{(U-Q)^2x}{x} = \Sp{(U-Q)y}{x} =0.$$ Zato je $(U-Q)x=0$ oziroma $Ux = Qx$. Ker ta sklep velja za vsak element $x\in X$, sledi, da je $U=Q$.
\end{dokaz}

Za konec tega pogljavja pokažimo, da je vsak obrnljiv operator na kompleksnem Hilbertovem prostoru produkt unitarnega in pozitivnega operatorja. Dekompozicijo obrnljivega operatorja na taka faktorja imenujemo ">polarna dekompozicija"<, saj spominja na polarni zapis kompleksnih števil.

\begin{izrek}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor. Tedaj za vsak obrnjiv operator $T\in B(X)$ obstajata tak unitaren operator $U$ na $X$ in tak pozitiven operator $R$ na $X$, da je $T = UR$.
\end{izrek}

\begin{dokaz}
	Naj bo $X$ poljuben kompleksen Hilbertov prostor in vzemimo poljuben obrnljiv operator $T\in B(X)$. Ker je operator $T$ obrnjiv, nam lema \ref{lem:adjinv} pove, da je tudi $T^*$ obrnljiv operator. Posledično je tudi operator $T^*T$ obrnjiv. Še več, lema \ref{lem:sebiadjlast} nam pove, da je operator $T^*T$ sebi-adjungiran. Vzemimo sedaj poljuben element $x\in X$. Tedaj je: $$\Sp{T^*Tx}{x} = \Sp{Tx}{Tx} = \norm{Tx}^2 \geq 0.$$ Ker sklep velja za poljuben element $x\in X$, sledi, da je operator $T^*T$ pozitivno semidefiniten. To pomeni, da je operator $T^*T$ pozitiven. Izrek \ref{izr:pozroot} nam sedaj pove, da operator $T^*T$ premore enolično določen pozitiven koren, ki ga označimo z $R$. Ker je operator $T^*T$ obrnljiv, nam opomba \ref{opomb:specter} pove, da je $0\neq\sigma(T^*T)$. Dodatno nam izrek \ref{izr:polyspec} pove, da je: $$\sigma(T^*T) = \{\lambda^2~|~\lambda\in\sigma(R)\}.$$ To pomeni, da $0\neq \sigma(R)$, torej je operator $R$ obrnljiv. Označimo sedaj operator $U = TR^{-1}$ in opazimo, da je obrnljiv. To pomeni, da je $\operatorname{Im}U = X$. Hkrati velja: $$U^*U = (R^{-1})^*T^*TR^{-1} = (R^*)^{-1}R^2R^{-1}= R^{-1}R = I.$$ Po točki $a)$ izreka \ref{izr:uniizo} je potem operator $U$ izometrija. Še več, operator $U$ je surjektivna izometrija. Zato je po točki $b)$ izreka \ref{izr:uniizo} opertor $U$ unitaren. 
\end{dokaz}
